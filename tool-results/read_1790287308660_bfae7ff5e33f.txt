     1→"use client";
     2→
     3→import { useCallback, useEffect, useMemo, useState } from "react";
     4→import Link from "next/link";
     5→import {
     6→  GitCompare,
     7→  Loader2,
     8→  Save,
     9→  CheckCircle2,
    10→  Trash2,
    11→  Search,
    12→  Sparkles,
    13→  Eye,
    14→  ExternalLink,
    15→  AlertCircle,
    16→  X,
    17→} from "lucide-react";
    18→import { useToast } from "@/hooks/use-toast";
    19→import { toFa, faDate } from "@/lib/format";
    20→
    21→/* ============================================================
    22→   /admin/compare — admin overview for the HEAVIX Machine
    23→   Comparison Engine (V1.0).
    24→   • Lists saved comparison sessions with item counts + AI summaries
    25→   • Configure which attributes are shown in the public compare table
    26→   • Drill into a session to view its full comparison + AI summary
    27→   ============================================================ */
    28→
    29→type Session = {
    30→  id: string;
    31→  name: string | null;
    32→  status: string;
    33→  userId: string | null;
    34→  shareToken: string | null;
    35→  createdAt: string;
    36→  updatedAt: string;
    37→  aiSummary: string | null;
    38→  aiSummaryAt: string | null;
    39→  itemCount: number;
    40→};
    41→
    42→type Attribute = {
    43→  id: string;
    44→  key: string | null;
    45→  name: string;
    46→  nameEn: string | null;
    47→  labelFa: string | null;
    48→  labelEn: string | null;
    49→  type: string;
    50→  unit: string | null;
    51→  sortOrder: number;
    52→};
    53→
    54→type SessionDetail = {
    55→  id: string;
    56→  name: string | null;
    57→  status: string;
    58→  userId: string | null;
    59→  shareToken: string | null;
    60→  shareExpiresAt: string | null;
    61→  createdAt: string;
    62→  updatedAt: string;
    63→  aiSummary: string | null;
    64→  aiSummaryAt: string | null;
    65→  items: {
    66→    id: string;
    67→    listingId: string | null;
    68→    productId: string | null;
    69→    brandId: string | null;
    70→    modelId: string | null;
    71→    sortOrder: number;
    72→  }[];
    73→};
    74→
    75→type Cell = {
    76→  itemId: string;
    77→  display: string;
    78→  raw: string | null;
    79→  unit?: string | null;
    80→  normalized: string;
    81→  provenance?: {
    82→    source?: string | null;
    83→    confidence?: number | null;
    84→    verifiedAt?: string | null;
    85→  };
    86→};
    87→
    88→type Row = {
    89→  key: string;
    90→  label: string;
    91→  category: "fixed" | "attribute";
    92→  unit?: string | null;
    93→  cells: Cell[];
    94→  isDifferent: boolean;
    95→  isPrice?: boolean;
    96→};
    97→
    98→type ItemData = {
    99→  id: string;
   100→  title: string;
   101→  slug?: string | null;
   102→  image?: string | null;
   103→  brandName?: string | null;
   104→  categoryName?: string | null;
   105→};
   106→
   107→type ComparisonData = {
   108→  items: ItemData[];
   109→  rows: Row[];
   110→  attributes: string[];
   111→  differences: string[];
   112→  crossCategoryWarning: boolean;
   113→  categories: { id: string; name: string }[];
   114→};
   115→
   116→export default function AdminComparePage() {
   117→  const { toast } = useToast();
   118→  const [loading, setLoading] = useState(true);
   119→  const [sessions, setSessions] = useState<Session[]>([]);
   120→  const [attributes, setAttributes] = useState<Attribute[]>([]);
   121→  const [visibleIds, setVisibleIds] = useState<string[]>([]);
   122→  const [savingAttrs, setSavingAttrs] = useState(false);
   123→  const [savedAttrs, setSavedAttrs] = useState(false);
   124→  const [attrFilter, setAttrFilter] = useState("");
   125→
   126→  // Session detail drawer
   127→  const [detailSession, setDetailSession] = useState<SessionDetail | null>(null);
   128→  const [detailData, setDetailData] = useState<ComparisonData | null>(null);
   129→  const [detailLoading, setDetailLoading] = useState(false);
   130→
   131→  // Renaming state
   132→  const [renameId, setRenameId] = useState<string | null>(null);
   133→  const [renameValue, setRenameValue] = useState("");
   134→
   135→  const load = useCallback(async () => {
   136→    setLoading(true);
   137→    try {
   138→      const res = await fetch("/api/admin/compare", { cache: "no-store" });
   139→      if (!res.ok) throw new Error("خطا در بارگذاری");
   140→      const json = await res.json();
   141→      setSessions(json.sessions ?? []);
   142→      setAttributes(json.attributes ?? []);
   143→      setVisibleIds(Array.isArray(json.visibleAttributeIds) ? json.visibleAttributeIds : []);
   144→    } catch (err: any) {
   145→      toast({
   146→        title: "خطا",
   147→        description: err?.message ?? "خطا در بارگذاری",
   148→        variant: "destructive",
   149→      });
   150→    } finally {
   151→      setLoading(false);
   152→    }
   153→  }, [toast]);
   154→
   155→  useEffect(() => {
   156→    load();
   157→  }, [load]);
   158→
   159→  const toggleAttr = (id: string) => {
   160→    setVisibleIds((prev) =>
   161→      prev.includes(id) ? prev.filter((x) => x !== id) : [...prev, id],
   162→    );
   163→  };
   164→
   165→  const saveAttrs = async () => {
   166→    setSavingAttrs(true);
   167→    try {
   168→      const res = await fetch("/api/admin/compare", {
   169→        method: "PUT",
   170→        headers: { "Content-Type": "application/json" },
   171→        body: JSON.stringify({ visibleAttributeIds: visibleIds }),
   172→      });
   173→      if (!res.ok) throw new Error("ذخیره ناموفق بود");
   174→      setSavedAttrs(true);
   175→      setTimeout(() => setSavedAttrs(false), 2500);
   176→      toast({ title: "ذخیره شد", description: "فهرست ویژگی‌های قابل نمایش به‌روزرسانی شد." });
   177→    } catch (err: any) {
   178→      toast({
   179→        title: "خطا",
   180→        description: err?.message ?? "ذخیره ناموفق بود",
   181→        variant: "destructive",
   182→      });
   183→    } finally {
   184→      setSavingAttrs(false);
   185→    }
   186→  };
   187→
   188→  const archiveSession = async (id: string) => {
   189→    if (!confirm("این مقایسه آرشیو شود؟")) return;
   190→    try {
   191→      const res = await fetch(`/api/admin/compare/${id}`, { method: "DELETE" });
   192→      if (!res.ok) throw new Error("خطا در آرشیو");
   193→      toast({ title: "آرشیو شد" });
   194→      load();
   195→    } catch (err: any) {
   196→      toast({
   197→        title: "خطا",
   198→        description: err?.message ?? "خطا در آرشیو",
   199→        variant: "destructive",
   200→      });
   201→    }
   202→  };
   203→
   204→  const renameSession = async (id: string) => {
   205→    try {
   206→      const res = await fetch(`/api/admin/compare/${id}`, {
   207→        method: "PATCH",
   208→        headers: { "Content-Type": "application/json" },
   209→        body: JSON.stringify({ name: renameValue }),
   210→      });
   211→      if (!res.ok) throw new Error("خطا در تغییر نام");
   212→      toast({ title: "نام تغییر کرد" });
   213→      setRenameId(null);
   214→      load();
   215→    } catch (err: any) {
   216→      toast({
   217→        title: "خطا",
   218→        description: err?.message ?? "خطا در تغییر نام",
   219→        variant: "destructive",
   220→      });
   221→    }
   222→  };
   223→
   224→  const openDetail = async (id: string) => {
   225→    setDetailLoading(true);
   226→    setDetailSession(null);
   227→    setDetailData(null);
   228→    try {
   229→      const res = await fetch(`/api/admin/compare/${id}`, { cache: "no-store" });
   230→      if (!res.ok) throw new Error("خطا در بارگذاری");
   231→      const json = await res.json();
   232→      setDetailSession(json.session);
   233→      setDetailData(json.data);
   234→    } catch (err: any) {
   235→      toast({
   236→        title: "خطا",
   237→        description: err?.message ?? "خطا در بارگذاری",
   238→        variant: "destructive",
   239→      });
   240→    } finally {
   241→      setDetailLoading(false);
   242→    }
   243→  };
   244→
   245→  const filteredAttributes = useMemo(() => {
   246→    if (!attrFilter.trim()) return attributes;
   247→    const q = attrFilter.trim().toLowerCase();
   248→    return attributes.filter(
   249→      (a) =>
   250→        (a.name ?? "").toLowerCase().includes(q) ||
   251→        (a.nameEn ?? "").toLowerCase().includes(q) ||
   252→        (a.key ?? "").toLowerCase().includes(q),
   253→    );
   254→  }, [attributes, attrFilter]);
   255→
   256→  const visibleCount = visibleIds.length;
   257→
   258→  if (loading) {
   259→    return (
   260→      <div className="flex h-64 items-center justify-center">
   261→        <Loader2 className="h-6 w-6 animate-spin text-[#F58220]" />
   262→      </div>
   263→    );
   264→  }
   265→
   266→  return (
   267→    <div className="space-y-6">
   268→      {/* Header */}
   269→      <div>
   270→        <h1 className="flex items-center gap-2 text-2xl font-black text-zinc-900">
   271→          <GitCompare className="h-6 w-6 text-[#F58220]" />
   272→          موتور مقایسه ماشین‌آلات
   273→        </h1>
   274→        <p className="mt-1 text-sm text-zinc-500">
   275→          مدیریت جلسه‌های مقایسهٔ ذخیره‌شده، پیکربندی ویژگی‌های قابل نمایش و مشاهدهٔ
   276→          خلاصه‌های هوش مصنوعی.
   277→        </p>
   278→      </div>
   279→
   280→      {/* Stats row */}
   281→      <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
   282→        <StatCard label="جلسه‌های فعال" value={sessions.filter((s) => s.status === "ACTIVE").length} />
   283→        <StatCard label="جلسه‌های آرشیو‌شده" value={sessions.filter((s) => s.status === "ARCHIVED").length} />
   284→        <StatCard label="خلاصه‌های AI تولیدشده" value={sessions.filter((s) => s.aiSummary).length} />
   285→        <StatCard label="ویژگی‌های قابل نمایش" value={visibleCount === 0 ? "همه" : visibleCount} />
   286→      </div>
   287→
   288→      {/* Sessions table */}
   289→      <div className="rounded-2xl border border-zinc-200 bg-white p-6 shadow-sm">
   290→        <h2 className="mb-4 text-lg font-black text-zinc-900">جلسه‌های مقایسه</h2>
   291→        {sessions.length === 0 ? (
   292→          <div className="rounded-xl border border-dashed border-zinc-200 p-8 text-center">
   293→            <AlertCircle className="mx-auto mb-3 h-8 w-8 text-zinc-300" />
   294→            <p className="text-sm text-zinc-500">هنوز جلسه‌ای ذخیره نشده است.</p>
   295→            <p className="mt-1 text-xs text-zinc-400">
   296→              کاربران با مراجعه به <Link href="/compare" className="text-[#F58220] hover:underline">/compare</Link> و
   297→              ذخیرهٔ مقایسه، جلسه‌ها را اینجا ایجاد می‌کنند.
   298→            </p>
   299→          </div>
   300→        ) : (
   301→          <div className="overflow-x-auto">
   302→            <table className="w-full min-w-[800px]">
   303→              <thead>
   304→                <tr className="border-b border-zinc-200 text-right text-xs font-bold text-zinc-500">
   305→                  <th className="px-3 py-3">نام</th>
   306→                  <th className="px-3 py-3">وضعیت</th>
   307→                  <th className="px-3 py-3">تعداد موارد</th>
   308→                  <th className="px-3 py-3">خلاصه AI</th>
   309→                  <th className="px-3 py-3">تاریخ ایجاد</th>
   310→                  <th className="px-3 py-3">آخرین به‌روزرسانی</th>
   311→                  <th className="px-3 py-3 text-left">عملیات</th>
   312→                </tr>
   313→              </thead>
   314→              <tbody>
   315→                {sessions.map((s) => (
   316→                  <tr key={s.id} className="border-b border-zinc-100 text-sm hover:bg-zinc-50">
   317→                    <td className="px-3 py-3">
   318→                      {renameId === s.id ? (
   319→                        <div className="flex items-center gap-1">
   320→                          <input
   321→                            type="text"
   322→                            value={renameValue}
   323→                            onChange={(e) => setRenameValue(e.target.value)}
   324→                            className="h-8 w-40 rounded-lg border border-zinc-200 px-2 text-xs"
   325→                            autoFocus
   326→                          />
   327→                          <button
   328→                            onClick={() => renameSession(s.id)}
   329→                            className="rounded bg-[#F58220] px-2 py-1 text-[10px] text-white"
   330→                          >
   331→                            ذخیره
   332→                          </button>
   333→                          <button
   334→                            onClick={() => setRenameId(null)}
   335→                            className="rounded px-1 text-zinc-400 hover:text-zinc-600"
   336→                          >
   337→                            <X className="h-3 w-3" />
   338→                          </button>
   339→                        </div>
   340→                      ) : (
   341→                        <button
   342→                          onClick={() => {
   343→                            setRenameId(s.id);
   344→                            setRenameValue(s.name ?? "");
   345→                          }}
   346→                          className="font-bold text-zinc-800 hover:text-[#F58220]"
   347→                          title="کلیک برای ویرایش نام"
   348→                        >
   349→                          {s.name ?? <span className="text-zinc-400">بدون نام</span>}
   350→                        </button>
   351→                      )}
   352→                    </td>
   353→                    <td className="px-3 py-3">
   354→                      <span
   355→                        className={`inline-flex rounded-full px-2 py-0.5 text-[10px] font-bold ${
   356→                          s.status === "ACTIVE"
   357→                            ? "bg-emerald-100 text-emerald-700"
   358→                            : "bg-zinc-100 text-zinc-500"
   359→                        }`}
   360→                      >
   361→                        {s.status === "ACTIVE" ? "فعال" : "آرشیو"}
   362→                      </span>
   363→                    </td>
   364→                    <td className="px-3 py-3 text-zinc-700">{toFa(s.itemCount)}</td>
   365→                    <td className="px-3 py-3">
   366→                      {s.aiSummary ? (
   367→                        <span className="inline-flex items-center gap-1 text-[10px] text-emerald-600">
   368→                          <Sparkles className="h-3 w-3" />
   369→                          {s.aiSummaryAt ? faDate(s.aiSummaryAt) : "دارد"}
   370→                        </span>
   371→                      ) : (
   372→                        <span className="text-[10px] text-zinc-400">—</span>
   373→                      )}
   374→                    </td>
   375→                    <td className="px-3 py-3 text-xs text-zinc-500">{faDate(s.createdAt)}</td>
   376→                    <td className="px-3 py-3 text-xs text-zinc-500">{faDate(s.updatedAt)}</td>
   377→                    <td className="px-3 py-3 text-left">
   378→                      <div className="flex justify-end gap-1">
   379→                        <button
   380→                          onClick={() => openDetail(s.id)}
   381→                          className="inline-flex items-center gap-1 rounded-lg border border-zinc-200 px-2 py-1 text-[10px] font-bold text-zinc-700 hover:border-[#F58220] hover:text-[#F58220]"
   382→                        >
   383→                          <Eye className="h-3 w-3" /> مشاهده
   384→                        </button>
   385→                        {s.shareToken && (
   386→                          <Link
   387→                            href={`/compare?share=${s.shareToken}`}
   388→                            target="_blank"
   389→                            rel="noopener noreferrer"
   390→                            className="inline-flex items-center gap-1 rounded-lg border border-zinc-200 px-2 py-1 text-[10px] font-bold text-zinc-700 hover:border-[#F58220] hover:text-[#F58220]"
   391→                          >
   392→                            <ExternalLink className="h-3 w-3" /> لینک
   393→                          </Link>
   394→                        )}
   395→                        <button
   396→                          onClick={() => archiveSession(s.id)}
   397→                          className="inline-flex items-center gap-1 rounded-lg border border-zinc-200 px-2 py-1 text-[10px] font-bold text-red-600 hover:border-red-300 hover:bg-red-50"
   398→                        >
   399→                          <Trash2 className="h-3 w-3" /> آرشیو
   400→                        </button>
   401→                      </div>
   402→                    </td>
   403→                  </tr>
   404→                ))}
   405→              </tbody>
   406→            </table>
   407→          </div>
   408→        )}
   409→      </div>
   410→
   411→      {/* Attribute visibility config */}
   412→      <div className="rounded-2xl border border-zinc-200 bg-white p-6 shadow-sm">
   413→        <div className="mb-4 flex flex-wrap items-start justify-between gap-3">
   414→          <div>
   415→            <h2 className="text-lg font-black text-zinc-900">ویژگی‌های قابل نمایش در جدول مقایسه</h2>
   416→            <p className="mt-1 text-xs text-zinc-500">
   417→              وقتی هیچ ویژگی‌ای انتخاب نشود، <span className="font-bold">همه</span> ویژگی‌ها نمایش داده می‌شوند.
   418→              وقتی حداقل یک ویژگی انتخاب شود، فقط ویژگی‌های انتخاب‌شده در جدول عمومی مقایسه ظاهر می‌شوند.
   419→            </p>
   420→          </div>
   421→          <button
   422→            onClick={saveAttrs}
   423→            disabled={savingAttrs}
   424→            className="inline-flex items-center gap-2 rounded-xl bg-[#F58220] px-4 py-2 text-sm font-bold text-white transition hover:bg-[#ff8c38] disabled:opacity-50"
   425→          >
   426→            {savingAttrs ? (
   427→              <Loader2 className="h-4 w-4 animate-spin" />
   428→            ) : savedAttrs ? (
   429→              <CheckCircle2 className="h-4 w-4" />
   430→            ) : (
   431→              <Save className="h-4 w-4" />
   432→            )}
   433→            {savingAttrs ? "در حال ذخیره..." : savedAttrs ? "ذخیره شد!" : "ذخیره"}
   434→          </button>
   435→        </div>
   436→
   437→        <div className="mb-4 flex items-center gap-2 rounded-xl border border-zinc-200 bg-white px-3">
   438→          <Search className="h-4 w-4 text-zinc-400" />
   439→          <input
   440→            type="text"
   441→            value={attrFilter}
   442→            onChange={(e) => setAttrFilter(e.target.value)}
   443→            placeholder="جستجوی ویژگی — نام، کلید یا نام انگلیسی..."
   444→            className="h-10 flex-1 bg-transparent text-sm outline-none"
   445→          />
   446→          {attrFilter && (
   447→            <button onClick={() => setAttrFilter("")} className="text-zinc-400 hover:text-zinc-600">
   448→              <X className="h-4 w-4" />
   449→            </button>
   450→          )}
   451→        </div>
   452→
   453→        {attributes.length === 0 ? (
   454→          <div className="rounded-xl border border-dashed border-zinc-200 p-6 text-center text-sm text-zinc-500">
   455→            هیچ ویژگی‌ای تعریف نشده است.
   456→          </div>
   457→        ) : (
   458→          <div className="max-h-96 overflow-y-auto rounded-xl border border-zinc-100">
   459→            <table className="w-full">
   460→              <thead className="sticky top-0 bg-zinc-50">
   461→                <tr className="border-b border-zinc-100 text-right text-xs font-bold text-zinc-500">
   462→                  <th className="w-12 px-3 py-2 text-center">
   463→                    <input
   464→                      type="checkbox"
   465→                      checked={
   466→                        visibleIds.length === 0
   467→                          ? false
   468→                          : filteredAttributes.every((a) => visibleIds.includes(a.id))
   469→                      }
   470→                      onChange={(e) => {
   471→                        if (e.target.checked) {
   472→                          setVisibleIds((prev) =>
   473→                            Array.from(new Set([...prev, ...filteredAttributes.map((a) => a.id)])),
   474→                          );
   475→                        } else {
   476→                          setVisibleIds((prev) =>
   477→                            prev.filter((id) => !filteredAttributes.some((a) => a.id === id)),
   478→                          );
   479→                        }
   480→                      }}
   481→                    />
   482→                  </th>
   483→                  <th className="px-3 py-2">نام</th>
   484→                  <th className="px-3 py-2">کلید</th>
   485→                  <th className="px-3 py-2">نوع</th>
   486→                  <th className="px-3 py-2">واحد</th>
   487→                </tr>
   488→              </thead>
   489→              <tbody>
   490→                {filteredAttributes.map((a) => {
   491→                  const checked = visibleIds.includes(a.id);
   492→                  return (
   493→                    <tr key={a.id} className="border-b border-zinc-50 text-sm hover:bg-zinc-50">
   494→                      <td className="px-3 py-2 text-center">
   495→                        <input
   496→                          type="checkbox"
   497→                          checked={checked}
   498→                          onChange={() => toggleAttr(a.id)}
   499→                        />
   500→                      </td>
   501→                      <td className="px-3 py-2 font-bold text-zinc-800">
   502→                        {a.labelFa ?? a.name}
   503→                        {a.nameEn && (
   504→                          <span className="mr-2 text-[10px] text-zinc-400">{a.nameEn}</span>
   505→                        )}
   506→                      </td>
   507→                      <td className="px-3 py-2 text-[11px] text-zinc-500">{a.key ?? "—"}</td>
   508→                      <td className="px-3 py-2 text-[11px] text-zinc-500">{a.type}</td>
   509→                      <td className="px-3 py-2 text-[11px] text-zinc-500">{a.unit ?? "—"}</td>
   510→                    </tr>
   511→                  );
   512→                })}
   513→              </tbody>
   514→            </table>
   515→          </div>
   516→        )}
   517→        <p className="mt-3 text-[10px] text-zinc-400">
   518→          {visibleIds.length === 0
   519→            ? `در حال حاضر همهٔ ${toFa(attributes.length)} ویژگی نمایش داده می‌شوند.`
   520→            : `${toFa(visibleIds.length)} ویژگی از ${toFa(attributes.length)} ویژگی انتخاب شده است.`}
   521→        </p>
   522→      </div>
   523→
   524→      {/* Detail drawer */}
   525→      {(detailLoading || detailSession) && (
   526→        <div
   527→          className="fixed inset-0 z-50 flex items-center justify-end bg-black/40"
   528→          onClick={() => {
   529→            setDetailSession(null);
   530→            setDetailData(null);
   531→          }}
   532→        >
   533→          <div
   534→            className="flex h-full w-full max-w-3xl flex-col bg-white shadow-2xl"
   535→            onClick={(e) => e.stopPropagation()}
   536→          >
   537→            <div className="flex items-center justify-between border-b border-zinc-200 px-6 py-4">
   538→              <div>
   539→                <h3 className="text-lg font-black text-zinc-900">
   540→                  {detailSession?.name ?? "جزئیات مقایسه"}
   541→                </h3>
   542→                <p className="text-xs text-zinc-500">
   543→                  {detailSession && faDate(detailSession.createdAt)} ·{" "}
   544→                  {detailSession && toFa(detailSession.items.length)} مورد
   545→                </p>
   546→              </div>
   547→              <button
   548→                onClick={() => {
   549→                  setDetailSession(null);
   550→                  setDetailData(null);
   551→                }}
   552→                className="rounded-lg p-2 text-zinc-400 hover:bg-zinc-100"
   553→              >
   554→                <X className="h-5 w-5" />
   555→              </button>
   556→            </div>
   557→
   558→            <div className="min-h-0 flex-1 overflow-y-auto p-6">
   559→              {detailLoading ? (
   560→                <div className="flex h-40 items-center justify-center">
   561→                  <Loader2 className="h-6 w-6 animate-spin text-[#F58220]" />
   562→                </div>
   563→              ) : detailSession && detailData ? (
   564→                <div className="space-y-6">
   565→                  {/* AI summary */}
   566→                  {detailSession.aiSummary && (
   567→                    <div className="rounded-xl border border-[#F58220]/30 bg-[#F58220]/5 p-4">
   568→                      <div className="mb-2 flex items-center gap-2">
   569→                        <Sparkles className="h-4 w-4 text-[#F58220]" />
   570→                        <h4 className="text-sm font-black text-zinc-800">خلاصه هوش مصنوعی</h4>
   571→                        {detailSession.aiSummaryAt && (
   572→                          <span className="text-[10px] text-zinc-500">
   573→                            ({faDate(detailSession.aiSummaryAt)})
   574→                          </span>
   575→                        )}
   576→                      </div>
   577→                      <div className="whitespace-pre-line text-xs leading-6 text-zinc-700">
   578→                        {detailSession.aiSummary}
   579→                      </div>
   580→                    </div>
   581→                  )}
   582→
   583→                  {/* Cross-category warning */}
   584→                  {detailData.crossCategoryWarning && detailData.categories.length > 1 && (
   585→                    <div className="rounded-xl border border-amber-200 bg-amber-50 p-3 text-xs text-amber-800">
   586→                      <AlertCircle className="ml-1 inline h-3 w-3" />
   587→                      موارد از دسته‌های متفاوت هستند:{" "}
   588→                      {detailData.categories.map((c) => c.name).join("، ")}
   589→                    </div>
   590→                  )}
   591→
   592→                  {/* Comparison table */}
   593→                  {detailData.items.length === 0 ? (
   594→                    <div className="rounded-xl border border-dashed border-zinc-200 p-6 text-center text-sm text-zinc-500">
   595→                      این جلسه هیچ مورد معتبری ندارد (ممکن است آگهی‌ها حذف شده باشند).
   596→                    </div>
   597→                  ) : (
   598→                    <div className="overflow-x-auto rounded-xl border border-zinc-100">
   599→                      <table className="w-full min-w-[600px]">
   600→                        <thead className="bg-zinc-50">
   601→                          <tr className="border-b border-zinc-100">
   602→                            <th className="w-32 px-3 py-2 text-right text-[10px] font-bold text-zinc-500">
   603→                              مشخصه
   604→                            </th>
   605→                            {detailData.items.map((it) => (
   606→                              <th key={it.id} className="px-3 py-2 text-center text-[11px] font-bold text-zinc-700">
   607→                                <Link
   608→                                  href={it.slug ? `/listings/${it.slug}` : "#"}
   609→                                  target="_blank"
   610→                                  className="hover:text-[#F58220]"
   611→                                >
   612→                                  {it.title}
   613→                                </Link>
   614→                              </th>
   615→                            ))}
   616→                          </tr>
   617→                        </thead>
   618→                        <tbody>
   619→                          {detailData.rows.map((row, idx) => (
   620→                            <tr key={row.key} className={idx % 2 === 0 ? "bg-white" : "bg-zinc-50/50"}>
   621→                              <td className="px-3 py-2 text-[11px] font-bold text-zinc-600">
   622→                                <span className="flex items-center gap-1">
   623→                                  {row.label}
   624→                                  {row.isDifferent && (
   625→                                    <span className="inline-block h-1.5 w-1.5 rounded-full bg-[#F58220]" />
   626→                                  )}
   627→                                </span>
   628→                              </td>
   629→                              {row.cells.map((cell) => (
   630→                                <td
   631→                                  key={cell.itemId}
   632→                                  className={`px-3 py-2 text-center text-[11px] ${
   633→                                    row.isPrice ? "font-bold text-[#F58220]" : "text-zinc-700"
   634→                                  }`}
   635→                                >
   636→                                  {cell.display || "—"}
   637→                                </td>
   638→                              ))}
   639→                            </tr>
   640→                          ))}
   641→                        </tbody>
   642→                      </table>
   643→                    </div>
   644→                  )}
   645→                </div>
   646→              ) : (
   647→                <div className="text-center text-sm text-zinc-500">خطا در بارگذاری جزئیات</div>
   648→              )}
   649→            </div>
   650→          </div>
   651→        </div>
   652→      )}
   653→    </div>
   654→  );
   655→}
   656→
   657→function StatCard({ label, value }: { label: string; value: number | string }) {
   658→  return (
   659→    <div className="rounded-2xl border border-zinc-200 bg-white p-4 shadow-sm">
   660→      <div className="text-2xl font-black text-[#F58220]">{value}</div>
   661→      <div className="mt-1 text-[11px] font-bold text-zinc-500">{label}</div>
   662→    </div>
   663→  );
   664→}
