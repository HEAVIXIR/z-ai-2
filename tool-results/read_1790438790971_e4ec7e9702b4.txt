   124→function listRouteFiles(): string[] {
   125→  const results: string[] = [];
   126→  function walk(dir: string) {
   127→    const entries = fs.readdirSync(dir, { withFileTypes: true });
   128→    for (const entry of entries) {
   129→      const full = path.join(dir, entry.name);
   130→      if (entry.isDirectory()) walk(full);
   131→      else if (entry.name === 'route.ts') results.push(full);
   132→    }
   133→  }
   134→  walk(STORE_ROUTES_DIR);
   135→  // Exclude health endpoints (monitoring stubs — no auth required)
   136→  return results.filter(f => !f.includes('/health/')).sort();
   137→}
   138→
   139→function readRoute(rel: string): string {
   140→  return fs.readFileSync(rel, 'utf8');
   141→}
   142→
   143→function listServiceFiles(): string[] {
   144→  return STORE_SERVICE_FILES.filter(f => fs.existsSync(f));
   145→}
   146→
   147→function readService(rel: string): string {
   148→  return fs.readFileSync(rel, 'utf8');
   149→}
   150→
   151→// Combined scan: returns content of every route file + every service file.
   152→// Used for audit/mutation/action-key contracts (which now live in both).
   153→function readAllStoreFiles(): { path: string; content: string }[] {
   154→  const routes = listRouteFiles().map(p => ({ path: p, content: readRoute(p) }));
   155→  const services = listServiceFiles().map(p => ({ path: p, content: readService(p) }));
   156→  return [...routes, ...services];
   157→}
