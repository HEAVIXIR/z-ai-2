   229→/* ─────────────────── EstimatesTab ─────────────────── */
   230→
   231→function EstimatesTab() {
   232→  const [query, setQuery] = useState("");
   233→  const [searching, setSearching] = useState(false);
   234→  const [results, setResults] = useState<
   235→    Array<{ id: string; title: string; slug: string; price: string | null; brand?: { name: string } | null; category?: { name: string } | null }>
   236→  >([]);
   237→  const [selectedId, setSelectedId] = useState<string | null>(null);
   238→  const [estimate, setEstimate] = useState<Estimate | null>(null);
   239→  const [health, setHealth] = useState<Health | null>(null);
   240→  const [loading, setLoading] = useState(false);
   241→  const [error, setError] = useState<string | null>(null);
   242→
   243→  // Debounced listing search
   244→  useEffect(() => {
   245→    const q = query.trim();
   246→    if (q.length < 2) {
   247→      setResults([]);
   248→      return;
   249→    }
   250→    let cancelled = false;
   251→    setSearching(true);
   252→    const t = setTimeout(async () => {
   253→      try {
   254→        const res = await fetch(
   255→          `/api/listings?q=${encodeURIComponent(q)}&limit=15`,
   256→        );
   257→        if (!res.ok) throw new Error("search failed");
   258→        const json = await res.json();
   259→        if (!cancelled) {
   260→          setResults(json.data ?? []);
   261→        }
   262→      } catch {
   263→        if (!cancelled) setResults([]);
   264→      } finally {
   265→        if (!cancelled) setSearching(false);
   266→      }
   267→    }, 350);
   268→    return () => {
   269→      cancelled = true;
   270→      clearTimeout(t);
   271→    };
   272→  }, [query]);
   273→
   274→  const loadEstimate = useCallback(async (listingId: string) => {
   275→    setSelectedId(listingId);
   276→    setEstimate(null);
   277→    setHealth(null);
   278→    setError(null);
   279→    setLoading(true);
   280→    try {
   281→      const [eRes, hRes] = await Promise.all([
   282→        fetch(`/api/pricing/estimate?listingId=${listingId}`),
   283→        fetch(`/api/pricing/health?listingId=${listingId}`),
   284→      ]);
   285→      if (!eRes.ok) throw new Error("estimate fetch failed");
   286→      const e = (await eRes.json()) as Estimate;
   287→      setEstimate(e);
   288→      if (hRes.ok) {
   289→        setHealth((await hRes.json()) as Health);
   290→      }
   291→    } catch (err: any) {
   292→      setError(err?.message ?? "خطا در دریافت تخمین قیمت");
   293→    } finally {
   294→      setLoading(false);
   295→    }
   296→  }, []);
   297→
   298→  return (
   299→    <div className="grid gap-6 lg:grid-cols-[360px_1fr]">
   300→      {/* Picker */}
   301→      <div className="rounded-2xl border border-zinc-200 bg-white p-5 shadow-sm">
   302→        <h2 className="mb-3 text-sm font-bold text-zinc-900">انتخاب آگهی</h2>
   303→        <input
   304→          type="text"
   305→          value={query}
   306→          onChange={(e) => setQuery(e.target.value)}
   307→          placeholder="جستجوی عنوان آگهی…"
   308→          className="w-full rounded-xl border border-zinc-200 px-4 py-2.5 text-sm outline-none focus:border-[#F58220]"
   309→        />
   310→        {searching && (
   311→          <div className="mt-3 flex items-center gap-2 text-xs text-zinc-400">
   312→            <Loader2 className="h-3.5 w-3.5 animate-spin" />
   313→            در حال جستجو…
   314→          </div>
   315→        )}
   316→        {!searching && results.length > 0 && (
   317→          <div className="mt-3 max-h-96 space-y-1.5 overflow-y-auto pl-1">
   318→            {results.map((r) => (
   319→              <button
   320→                key={r.id}
   321→                type="button"
   322→                onClick={() => loadEstimate(r.id)}
   323→                className={`block w-full rounded-xl border px-3 py-2 text-right transition ${
   324→                  selectedId === r.id
   325→                    ? "border-[#F58220] bg-[#F58220]/10"
   326→                    : "border-zinc-200 hover:border-zinc-300 hover:bg-zinc-50"
   327→                }`}
   328→              >
   329→                <div className="truncate text-sm font-semibold text-zinc-800">
   330→                  {r.title}
   331→                </div>
   332→                <div className="mt-0.5 truncate text-[11px] text-zinc-500">
   333→                  {r.brand?.name ?? "—"} · {r.category?.name ?? "—"}
   334→                  {r.price != null && ` · ${formatCompactPrice(BigInt(r.price))}`}
   335→                </div>
   336→              </button>
   337→            ))}
   338→          </div>
   339→        )}
   340→        {!searching && query.trim().length >= 2 && results.length === 0 && (
   341→          <div className="mt-3 text-xs text-zinc-400">
   342→            نتیجه‌ای یافت نشد.
   343→          </div>
   344→        )}
   345→      </div>
   346→
   347→      {/* Estimate panel */}
   348→      <div className="rounded-2xl border border-zinc-200 bg-white p-6 shadow-sm">
   349→        {!selectedId && (
   350→          <div className="flex h-full flex-col items-center justify-center gap-3 py-16 text-center">
   351→            <Info className="h-8 w-8 text-zinc-300" />
   352→            <p className="text-sm text-zinc-400">
   353→              برای مشاهده تخمین قیمت، یک آگهی از ستون کنار انتخاب کنید.
   354→            </p>
   355→          </div>
   356→        )}
   357→
   358→        {selectedId && loading && (
   359→          <div className="flex items-center gap-2 py-16 text-sm text-zinc-400">
   360→            <Loader2 className="h-4 w-4 animate-spin" />
   361→            در حال محاسبه تخمین قیمت…
   362→          </div>
   363→        )}
   364→
   365→        {selectedId && !loading && error && (
   366→          <div className="rounded-xl border border-rose-200 bg-rose-50 p-4 text-sm text-rose-700">
   367→            {error}
   368→          </div>
   369→        )}
   370→
   371→        {selectedId && !loading && estimate && (
   372→          <div className="space-y-5">
   373→            {/* Headline */}
   374→            <div>
   375→              <div className="text-xs text-zinc-500">قیمت تخمینی HEAVIX</div>
   376→              {estimate.estimatedPrice != null ? (
   377→                <div className="mt-1 text-3xl font-black text-[#F58220]">
   378→                  {formatFullPrice(estimate.estimatedPrice)}
   379→                </div>
   380→              ) : (
   381→                <div className="mt-1 text-xl font-bold text-zinc-500">
   382→                  داده ناکافی برای تخمین
   383→                </div>
   384→              )}
   385→              {estimate.priceLower != null && estimate.priceUpper != null && (
   386→                <div className="mt-1 text-sm text-zinc-600">
   387→                  بازه: {formatCompactPrice(estimate.priceLower)} تا{" "}
   388→                  {formatCompactPrice(estimate.priceUpper)}
   389→                </div>
   390→              )}
   391→            </div>
   392→
   393→            {/* Confidence + freshness + count */}
   394→            <div className="grid grid-cols-3 gap-3">
   395→              <Stat
   396→                label="اطمینان"
   397→                value={
   398→                  CONFIDENCE_LABEL[estimate.confidence] ?? estimate.confidence
   399→                }
   400→                accent={
   401→                  estimate.confidence === "HIGH"
   402→                    ? "emerald"
   403→                    : estimate.confidence === "MEDIUM"
   404→                      ? "amber"
   405→                      : estimate.confidence === "LOW"
   406→                        ? "orange"
   407→                        : "zinc"
   408→                }
   409→              />
   410→              <Stat
   411→                label="موارد مشابه"
   412→                value={toFa(estimate.comparableCount)}
   413→              />
   414→              <Stat
   415→                label="تازگی داده"
   416→                value={
   417→                  estimate.dataFreshness
   418→                    ? FRESHNESS_LABEL[estimate.dataFreshness] ??
   419→                      estimate.dataFreshness
   420→                    : "—"
   421→                }
   422→              />
   423→            </div>
   424→
   425→            {/* Price health */}
   426→            {health && (
   427→              <div
   428→                className={`rounded-xl border p-4 ${HEALTH_COLOR[health.status] ?? "border-zinc-200 bg-zinc-50"}`}
   429→              >
   430→                <div className="flex items-center justify-between gap-3">
   431→                  <div className="flex items-center gap-2">
   432→                    <ShieldCheck className="h-5 w-5" />
   433→                    <span className="text-sm font-bold">
   434→                      وضعیت قیمت آگهی
   435→                    </span>
   436→                  </div>
   437→                  <span className="text-sm font-black">
   438→                    {HEALTH_LABEL[health.status] ?? health.status}
   439→                  </span>
   440→                </div>
   441→                {health.askingPrice != null && (
   442→                  <div className="mt-2 text-xs opacity-80">
   443→                    قیمت آگهی: {formatFullPrice(health.askingPrice)}
   444→                    {health.deviationPct != null && (
   445→                      <span className="mr-2">
   446→                        انحراف از میانه:{" "}
   447→                        {toFa((health.deviationPct * 100).toFixed(0))}٪
   448→                      </span>
   449→                    )}
   450→                  </div>
   451→                )}
   452→              </div>
   453→            )}
   454→
   455→            {/* Drivers */}
   456→            {estimate.mainDrivers.length > 0 && (
   457→              <div>
   458→                <div className="mb-2 flex items-center gap-2 text-xs font-bold text-zinc-500">
   459→                  <TrendingUp className="h-3.5 w-3.5" />
   460→                  عوامل مؤثر بر تخمین
   461→                </div>
   462→                <div className="flex flex-wrap gap-2">
   463→                  {estimate.mainDrivers.map((d, i) => (
   464→                    <span
   465→                      key={i}
   466→                      className="rounded-full bg-zinc-100 px-3 py-1 text-xs text-zinc-700"
   467→                    >
   468→                      {d}
   469→                    </span>
   470→                  ))}
   471→                </div>
   472→              </div>
   473→            )}
   474→
   475→            {/* Warnings */}
   476→            {estimate.warnings.length > 0 && (
   477→              <div className="space-y-2">
   478→                {estimate.warnings.map((w, i) => (
   479→                  <div
   480→                    key={i}
   481→                    className="flex items-start gap-2 rounded-xl border border-amber-200 bg-amber-50 p-3 text-xs text-amber-800"
   482→                  >
   483→                    <AlertTriangle className="mt-0.5 h-4 w-4 shrink-0" />
   484→                    <span>{w}</span>
   485→                  </div>
   486→                ))}
   487→              </div>
   488→            )}
   489→
   490→            {/* Comparables */}
   491→            {estimate.comparables.length > 0 && (
   492→              <div>
   493→                <div className="mb-2 flex items-center gap-2 text-xs font-bold text-zinc-500">
   494→                  <ListChecks className="h-3.5 w-3.5" />
   495→                  موارد مشابه ({toFa(estimate.comparables.length)} مورد)
   496→                </div>
   497→                <div className="max-h-72 space-y-1.5 overflow-y-auto pl-1">
   498→                  {estimate.comparables.map((c) => (
   499→                    <div
   500→                      key={c.id}
   501→                      className="flex items-center justify-between gap-3 rounded-xl border border-zinc-200 px-3 py-2"
   502→                    >
   503→                      <div className="min-w-0 flex-1">
   504→                        <div className="truncate text-sm font-semibold text-zinc-800">
   505→                          {c.slug ? (
   506→                            <a
   507→                              href={`/listings/${c.slug}`}
   508→                              target="_blank"
   509→                              rel="noopener noreferrer"
   510→                              className="hover:text-[#F58220]"
   511→                            >
   512→                              {c.title}
   513→                            </a>
   514→                          ) : (
   515→                            c.title
   516→                          )}
   517→                        </div>
   518→                        <div className="mt-0.5 text-[11px] text-zinc-500">
   519→                          {c.year ? `سال ${toFa(c.year)}` : "سال نامشخص"}
   520→                          {c.workingHours != null
   521→                            ? ` · ${toFa(c.workingHours)} ساعت`
   522→                            : ""}
   523→                          {c.city ? ` · ${c.city}` : ""}
   524→                        </div>
   525→                      </div>
   526→                      <div className="text-left">
   527→                        <div className="text-sm font-bold text-zinc-900">
   528→                          {formatCompactPrice(c.price)}
   529→                        </div>
   530→                        <div className="text-[10px] text-zinc-400">
   531→                          شباهت: {toFa((c.similarity * 100).toFixed(0))}٪
   532→                        </div>
   533→                      </div>
   534→                    </div>
   535→                  ))}
   536→                </div>
   537→              </div>
   538→            )}
   539→
   540→            {/* Disclaimer */}
   541→            <div className="rounded-xl border border-zinc-200 bg-zinc-50 p-3 text-[11px] leading-6 text-zinc-500">
   542→              {estimate.disclaimer ??
   543→                "این برآورد داده‌محور است و جایگزین کارشناسی حضوری نیست."}
   544→            </div>
   545→          </div>
   546→        )}
   547→      </div>
   548→    </div>
   549→  );
   550→}
