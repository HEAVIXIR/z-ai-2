    30→   docs/HEAVIX-MACHINE-COMPARISON-SPEC-V1.0.md
    31→
    32→   Flow:
    33→     • If ?share=TOKEN is present → load the shared session
    34→       (read-only) and render it.
    35→     • Otherwise → client-side session:
    36→         - Search listings via /api/listings
    37→         - Add → POST /api/compare (creates session lazily on
    38→           first add), then POST /api/compare/[id]/items
    39→         - Remove → DELETE /api/compare/[id]/items/[itemId]
    40→         - "تفاوت‌ها فقط" toggle (rows where isDifferent = true)
    41→         - "خلاصه هوش مصنوعی" → POST /api/compare/[id]/ai-summary
    42→         - "ذخیره مقایسه" → PATCH /api/compare/[id] { name }
    43→         - "اشتراک‌گذاری" → copies ${origin}/compare?share=TOKEN
    44→
    45→   Min 2 items, max 5 items. Dark theme, RTL, responsive.
    46→   ============================================================ */

   180→function ComparePageInner() {
   181→  const router = useRouter();
   182→  const searchParams = useSearchParams();
   183→  const shareToken = searchParams.get("share");
   184→  const isReadOnly = Boolean(shareToken);
   185→  const { toast } = useToast();
   186→
   187→  const [sessionId, setSessionId] = useState<string | null>(null);
   188→  const [session, setSession] = useState<Session | null>(null);
   189→  const [data, setData] = useState<ComparisonData | null>(null);
   190→  const [loading, setLoading] = useState(true);
   191→  const [differencesOnly, setDifferencesOnly] = useState(false);
   192→  const [aiSummary, setAiSummary] = useState<string | null>(null);
   193→  const [aiLoading, setAiLoading] = useState(false);
   194→  const [saving, setSaving] = useState(false);
   195→  const [shareCopied, setShareCopied] = useState(false);
   196→  const [saveName, setSaveName] = useState("");
   197→  const [showSaveDialog, setShowSaveDialog] = useState(false);
   198→
   199→  // ── Search state ──
   200→  const [search, setSearch] = useState("");
   201→  const [searchResults, setSearchResults] = useState<SearchResult[]>([]);
   202→  const [searching, setSearching] = useState(false);
   203→  const [showResults, setShowResults] = useState(false);
   204→  const searchDebounce = useRef<NodeJS.Timeout | null>(null);
   205→
   206→  /* ── Load session (shared or local) ── */
   207→  const loadSession = useCallback(async (id: string) => {
   208→    setLoading(true);
   209→    try {
   210→      const res = await fetch(`/api/compare/${id}`, { cache: "no-store" });
   211→      if (res.status === 404 || res.status === 410) {
   212→        // Session gone — drop the stale localStorage id and start fresh.
   213→        if (typeof window !== "undefined") localStorage.removeItem(SESSION_KEY);
   214→        setSessionId(null);
   215→        setSession(null);
   216→        setData(null);
   217→        setLoading(false);
   218→        return;
   219→      }
   220→      if (!res.ok) throw new Error("خطا در بارگذاری مقایسه");
   221→      const json = await res.json();
   222→      setSession(json.session);
   223→      setData(json.data);
   224→      setAiSummary(json.session?.aiSummary ?? null);
   225→    } catch (err: any) {
   226→      toast({
   227→        title: "خطا",
   228→        description: err?.message ?? "خطا در بارگذاری مقایسه",
   229→        variant: "destructive",
   230→      });
   231→    } finally {
   232→      setLoading(false);
   233→    }
   234→  }, [toast]);
   235→
   236→  const loadShared = useCallback(async (token: string) => {
   237→    setLoading(true);
   238→    try {
   239→      const res = await fetch(`/api/compare/shared/${token}`, { cache: "no-store" });
   240→      if (!res.ok) {
   241→        toast({
   242→          title: "لینک نامعتبر",
   243→          description: "لینک اشتراک‌گذاری نامعتبر یا منقضی است.",
   244→          variant: "destructive",
   245→        });
   246→        setLoading(false);
   247→        return;
   248→      }
   249→      const json = await res.json();
   250→      setSession({
   251→        id: json.session.id,
   252→        name: json.session.name,
   253→        shareToken: token,
   254→        shareExpiresAt: json.session.shareExpiresAt,
   255→        aiSummary: null,
   256→        aiSummaryAt: null,
   257→      });
   258→      setData(json.data);
   259→      setAiSummary(null);
   260→    } catch {
   261→      toast({
   262→        title: "خطا",
   263→        description: "خطا در بارگذاری مقایسهٔ اشتراک‌گذاری‌شده",
   264→        variant: "destructive",
   265→      });
   266→    } finally {
   267→      setLoading(false);
   268→    }
   269→  }, [toast]);
   270→
   271→  useEffect(() => {
   272→    if (shareToken) {
   273→      loadShared(shareToken);
   274→      return;
   275→    }
   276→    // Pick up pending listing IDs (set by the Compare button on the
   277→    // listing detail page) and add them to the session before rendering.
   278→    let pendingIds: string[] = [];
   279→    if (typeof window !== "undefined") {
   280→      try {
   281→        const raw = localStorage.getItem(PENDING_KEY);
   282→        if (raw) {
   283→          pendingIds = JSON.parse(raw);
   284→          if (!Array.isArray(pendingIds)) pendingIds = [];
   285→          pendingIds = pendingIds.filter((x) => typeof x === "string").slice(0, MAX_ITEMS);
   286→          localStorage.removeItem(PENDING_KEY);
   287→        }
   288→      } catch {
   289→        /* ignore */
   290→      }
   291→    }
   292→
   293→    // Try to resume an existing local session.
   294→    const savedId =
   295→      typeof window !== "undefined" ? localStorage.getItem(SESSION_KEY) : null;
   296→
   297→    if (pendingIds.length > 0) {
   298→      // Create a fresh session with the pending listings pre-populated.
   299→      (async () => {
   300→        setLoading(true);
   301→        try {
   302→          const res = await fetch("/api/compare", {
   303→            method: "POST",
   304→            headers: { "Content-Type": "application/json" },
   305→            body: JSON.stringify({ name: null, listingIds: pendingIds }),
   306→          });
   307→          if (!res.ok) throw new Error("خطا در ایجاد جلسهٔ مقایسه");
   308→          const json = await res.json();
   309→          const newId = json.session.id as string;
   310→          setSessionId(newId);
   311→          setSession(json.session);
   312→          if (typeof window !== "undefined") {
   313→            localStorage.setItem(SESSION_KEY, newId);
   314→          }
   315→          await loadSession(newId);
   316→        } catch {
   317→          setLoading(false);
   318→        }
   319→      })();
   320→      return;
   321→    }
   322→
   323→    if (savedId) {
   324→      loadSession(savedId);
   325→    } else {
   326→      setLoading(false);
   327→    }
   328→  }, [shareToken, loadShared, loadSession]);
   329→
   330→  /* ── Search ── */
   331→  const doSearch = useCallback(
   332→    (q: string) => {
   333→      setSearch(q);
   334→      if (searchDebounce.current) clearTimeout(searchDebounce.current);
   335→      if (q.trim().length < 2) {
   336→        setSearchResults([]);
   337→        setShowResults(false);
   338→        return;
   339→      }
   340→      setSearching(true);
   341→      searchDebounce.current = setTimeout(async () => {
   342→        try {
   343→          const res = await fetch(`/api/listings?q=${encodeURIComponent(q)}&limit=10`);
   344→          const json = await res.json();
   345→          if (json.success !== false) {
   346→            setSearchResults((json.data || json.listings || []).slice(0, 8));
   347→            setShowResults(true);
   348→          }
   349→        } catch {
   350→          /* ignore */
   351→        } finally {
   352→          setSearching(false);
   353→        }
   354→      }, 250);
   355→    },
   356→    [],
   357→  );
   358→
   359→  /* ── Add listing (creates session lazily on first add) ── */
   360→  const ensureSession = useCallback(async (): Promise<string | null> => {
   361→    if (sessionId) return sessionId;
   362→    try {
   363→      const res = await fetch("/api/compare", {
   364→        method: "POST",
   365→        headers: { "Content-Type": "application/json" },
   366→        body: JSON.stringify({ name: null }),
   367→      });
   368→      if (!res.ok) throw new Error("خطا در ایجاد جلسهٔ مقایسه");
   369→      const json = await res.json();
   370→      const newId = json.session.id as string;
   371→      setSessionId(newId);
   372→      setSession(json.session);
   373→      if (typeof window !== "undefined") {
   374→        localStorage.setItem(SESSION_KEY, newId);
   375→      }
   376→      return newId;
   377→    } catch (err: any) {
   378→      toast({
   379→        title: "خطا",
   380→        description: err?.message ?? "خطا در ایجاد جلسهٔ مقایسه",
   381→        variant: "destructive",
   382→      });
   383→      return null;
   384→    }
   385→  }, [sessionId, toast]);
   386→
   387→  const addListing = useCallback(
   388→    async (l: SearchResult) => {
   389→      // Cap check
   390→      if ((data?.items.length ?? 0) >= MAX_ITEMS) {
   391→        toast({
   392→          title: "حداکثر موارد",
   393→          description: `حداکثر ${toFa(MAX_ITEMS)} مورد قابل مقایسه است.`,
   394→          variant: "destructive",
   395→        });
   396→        return;
   397→      }
   398→      // Dedupe
   399→      if (data?.items.some((i) => i.listingId === l.id)) {
   400→        toast({ title: "تکراری", description: "این آگهی قبلاً اضافه شده است." });
   401→        return;
   402→      }
   403→      const id = await ensureSession();
   404→      if (!id) return;
   405→      try {
   406→        const res = await fetch(`/api/compare/${id}/items`, {
   407→          method: "POST",
   408→          headers: { "Content-Type": "application/json" },
   409→          body: JSON.stringify({ listingId: l.id }),
   410→        });
   411→        if (!res.ok) {
   412→          const j = await res.json().catch(() => ({}));
   413→          throw new Error(j.error ?? "خطا در افزودن مورد");
   414→        }
   415→        setSearch("");
   416→        setSearchResults([]);
   417→        setShowResults(false);
   418→        await loadSession(id);
   419→      } catch (err: any) {
   420→        toast({
   421→          title: "خطا",
   422→          description: err?.message ?? "خطا در افزودن مورد",
   423→          variant: "destructive",
   424→        });
   425→      }
   426→    },
   427→    [data, ensureSession, loadSession, toast],

   434→        await fetch(`/api/compare/${sessionId}/items/${itemId}`, {

   438→      } catch (err: any) {

   440→          title: "خطا",
   441→          description: err?.message ?? "خطا در حذف مورد",

   465→        title: "خطا",
   466→        description: err?.message ?? "خطا در تولید خلاصهٔ هوش مصنوعی",

   489→      setSession((prev) =>
   490→        prev
   491→          ? {
   492→              ...prev,
   493→              name: json.session.name,
   494→              shareToken: json.session.shareToken,
   495→              shareExpiresAt: json.session.shareExpiresAt,
   496→            }
   497→          : prev,
   498→      );

   501→    } catch (err: any) {
   502→      toast({
   503→        title: "خطا",
   504→        description: err?.message ?? "خطا در ذخیره",
   505→        variant: "destructive",
   506→      });
   507→    } finally {
   508→      setSaving(false);
   509→    }
   510→  }, [sessionId, saveName, session, toast]);
   511→
   512→  /* ── Share ── */
   513→  const shareLink = useMemo(() => {
   514→    if (!session?.shareToken) return null;
   515→    const origin = typeof window !== "undefined" ? window.location.origin : "";
   516→    return `${origin}/compare?share=${session.shareToken}`;
   517→  }, [session?.shareToken]);
   518→
   519→  const copyShareLink = useCallback(async () => {
   520→    if (!shareLink) return;
   521→    try {
   522→      await navigator.clipboard.writeText(shareLink);
   523→      setShareCopied(true);
   524→      setTimeout(() => setShareCopied(false), 2500);

   526→    } catch {
   527→      toast({
   528→        title: "خطا",
   529→        description: "کپی لینک ناموفق بود.",
   530→        variant: "destructive",
   531→      });
   532→    }
   533→  }, [shareLink, toast]);
   534→
   535→  /* ── Reset / new comparison ── */
   536→  const startNew = useCallback(() => {
   537→    if (typeof window !== "undefined") localStorage.removeItem(SESSION_KEY);
   538→    setSessionId(null);
   539→    setSession(null);
   540→    setData(null);
   541→    setAiSummary(null);
   542→    setSearch("");
   543→    setSearchResults([]);
   544→    setShowResults(false);
   545→    router.push("/compare");
   546→  }, [router]);
   547→
   548→  /* ── Filtered rows (differences-only mode) ── */
   549→  const visibleRows = useMemo(() => {
   550→    if (!data) return [];
   551→    if (!differencesOnly) return data.rows;
   552→    return data.rows.filter((r) => r.isDifferent);
   553→  }, [data, differencesOnly]);
   554→
   555→  const itemCount = data?.items.length ?? 0;
   556→  const canCompare = itemCount >= MIN_ITEMS;
   557→  const canSave = itemCount >= MIN_ITEMS && !isReadOnly;
   558→
   559→  /* ── Render ── */
   560→  return (
   561→    <div className="site-theme flex min-h-screen flex-col bg-[#0b0b0b]">
   562→      <Header categories={[]} />
   563→      <main className="flex-1 pt-28">
   564→        <div className="mx-auto max-w-[1400px] px-4 py-8 sm:px-6 lg:px-10 lg:py-12">
   565→          {/* ── Header ── */}
   566→          <div className="mb-8 flex flex-col items-start justify-between gap-4 lg:flex-row lg:items-end">
   567→            <div>
   568→              <div className="mb-3 inline-flex h-14 w-14 items-center justify-center rounded-2xl bg-[#F58220]/15">
   569→                <GitCompare className="h-7 w-7 text-[#F58220]" />

   571→              <h1 className="text-3xl font-black text-white lg:text-4xl">
   572→                {isReadOnly ? "مقایسهٔ اشتراک‌گذاری‌شده" : "موتور مقایسه ماشین‌آلات"}

   574→              <p className="mt-2 text-sm text-white/50">
   575→                {toFa(MIN_ITEMS)} تا {toFa(MAX_ITEMS)} آگهی را کنار هم بگذارید و تفاوت‌ها را ببینید.
   576→              </p>
