     5→/* ============================================================
     6→   HEAVIX — Machine Comparison Engine (V1.0)
     7→   ------------------------------------------------------------
     8→   docs/HEAVIX-MACHINE-COMPARISON-SPEC-V1.0.md
     9→
    10→   This engine turns 2-5 catalog/listing entities into a
    11→   structured comparison table:
    12→     • rows = attributes (fixed specs + dynamic ListingAttributeValue)
    13→     • columns = items (listings / products / models / brands)
    14→     • per-row "isDifferent" flag (drives Differences-Only mode)
    15→     • cross-category warning when items span >1 category
    16→     • price row carries the asking price + HEAVIX estimated range
    17→     • AI summary (no winner declaration) via z-ai-web-dev-sdk
    18→
    19→   Design rules (from the spec):
    20→     • Catalog-first: comparing a model ≠ comparing a listing.
    21→       The engine accepts either; the row builder resolves the
    22→       most specific data available.
    23→     • Missing data is shown as "—" (Not specified), NEVER as 0.
    24→     • Unit normalization: original value + normalized value + unit.
    25→       For V1 we keep it simple — we surface the unit and the
    26→       Persian-formatted display string.
    27→     • Provenance: each cell carries sourceType / confidence /
    28→       verifiedAt from ListingAttributeValue when available.
    29→     • No winner / no aggregate ranking — only difference flags
    30→       per row (Specification / Cost / Application / Condition
    31→       differences, per spec §12).
    32→
    33→   The module is intentionally server-only (imports Prisma).
    34→   ============================================================ */
    35→

   152→/** Build a random share token (URL-safe, 24 chars). */
   153→function generateShareToken(): string {
   154→  // 18 bytes → 24 base64url chars.
   155→  const bytes = new Uint8Array(18);
   156→  crypto.getRandomValues(bytes);
   157→  const b64 = btoa(String.fromCharCode(...bytes));
   158→  return b64.replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
   159→}
   160→
   161→/** Format a Toman price to Persian digits + suffix. */
   162→function formatToman(n: number | bigint | null | undefined): string {
   163→  if (n === null || n === undefined) return "";
   164→  const num = typeof n === "bigint" ? Number(n) : n;
   165→  if (!isFinite(num) || num <= 0) return "";
   166→  return `${num.toLocaleString("fa-IR")} تومان`;
   167→}
   168→
   169→/** Persian-formatted integer. */
   170→function formatInt(n: number | null | undefined): string {
   171→  if (n === null || n === undefined) return "";
   172→  return toFa(n);
   173→}

   179→/**
   180→ * Create a new comparison session with an optional share token.
   181→ * When `userId` is provided the session is owned by that user;
   182→ * otherwise it's anonymous (still usable via the public API).
   183→ */
   184→export async function createSession(
   185→  userId?: string | null,
   186→  name?: string,
   187→): Promise<{
   188→  id: string;
   189→  shareToken: string | null;
   190→  name: string | null;
   191→  status: string;
   192→  createdAt: Date;
   193→}> {
   194→  const session = await db.comparisonSession.create({
   195→    data: {
   196→      userId: userId ?? null,
   197→      name: name ?? null,
   198→      shareToken: generateShareToken(),
   199→      status: "ACTIVE",
   200→    },
   201→    select: {
   202→      id: true,
   203→      shareToken: true,
   204→      name: true,
   205→      status: true,
   206→      createdAt: true,
   207→    },
   208→  });
   209→  return session;
   210→}

   212→/**
   213→ * Add an item to a comparison session.
   214→ * Enforces a hard cap of 5 items per session and dedupes by listingId.
   215→ */
   216→export async function addItem(
   217→  sessionId: string,
   218→  input: AddItemInput,
   219→): Promise<{ id: string; sessionId: string; sortOrder: number }> {
   220→  // Verify session exists and is ACTIVE.
   221→  const session = await db.comparisonSession.findUnique({
   222→    where: { id: sessionId },
   223→    select: { id: true, status: true, items: { select: { id: true, listingId: true } } },
   224→  });
   225→  if (!session) throw new Error("Session not found");
   226→
   227→  // Cap at 5 items.
   228→  if (session.items.length >= 5) {
   229→    throw new Error("حداکثر ۵ مورد قابل مقایسه است");
   230→  }
   231→
   232→  // Dedupe by listingId (only when listingId provided).
   233→  if (input.listingId) {
   234→    const dup = session.items.find((i) => i.listingId === input.listingId);
   235→    if (dup) return { id: dup.id, sessionId, sortOrder: 0 };
   236→  }
   237→
   238→  const nextSort = session.items.length;
   239→  const item = await db.comparisonItem.create({
   240→    data: {
   241→      sessionId,
   242→      listingId: input.listingId ?? null,
   243→      productId: input.productId ?? null,
   244→      brandId: input.brandId ?? null,
   245→      modelId: input.modelId ?? null,
   246→      sortOrder: nextSort,
   247→    },
   248→    select: { id: true, sessionId: true, sortOrder: true },
   249→  });
   250→  return item;
   251→}
   252→
   253→/**
   254→ * Remove an item from a session.
   255→ */
   256→export async function removeItem(sessionId: string, itemId: string): Promise<void> {
   257→  await db.comparisonItem.deleteMany({ where: { id: itemId, sessionId } });
   258→}

   260→// ────────────────────────────────────────────────────────────
   261→// 2. Comparison data builder
   262→// ────────────────────────────────────────────────────────────
   263→
   264→/**
   265→ * Fetch + project all comparison items into a structured table.
   266→ *
   267→ * Returns:
   268→ *   items[]     — column metadata (one per ComparisonItem, in sortOrder)
   269→ *   rows[]      — row data (fixed specs first, then dynamic attributes)
   270→ *   attributes  — row keys in order
   271→ *   differences — row keys where isDifferent = true
   272→ *   crossCategoryWarning — true when items span >1 category
   273→ *   categories  — distinct categories present in the comparison
   274→ */
   275→export async function getComparisonData(sessionId: string): Promise<ComparisonData> {
   276→  const session = await db.comparisonSession.findUnique({
   277→    where: { id: sessionId },
   278→    select: {
   279→      id: true,
   280→      items: {
   281→        orderBy: { sortOrder: "asc" },
   282→        include: {
   283→          listing: {
   284→            include: {
   285→              brand: { select: { id: true, name: true, nameEn: true, country: true } },
   286→              category: { select: { id: true, name: true, slug: true, icon: true } },
   287→              model: { select: { id: true, name: true, nameEn: true } },
   288→              images: { orderBy: [{ isPrimary: "desc" }, { sortOrder: "asc" }], take: 1 },
   289→            },
   290→          },
   291→        },
   292→      },
   293→    },
   294→  });
   295→  if (!session) throw new Error("Session not found");
   296→
   297→  // ── Column metadata: only items that actually have a listing or
   298→  // a brand/model/product to show. Items with no underlying entity
   299→  // (e.g. listingId pointing to a deleted listing) are skipped —
   300→  // the cascade SetNull on Listing.comparisonItems means listingId
   301→  // becomes null but the ComparisonItem row remains, so filter. ──
   302→  const items: ComparisonItemData[] = [];
   303→  const listingIds: string[] = [];
   304→  for (const it of session.items) {
   305→    if (it.listing) {
   306→      const l = it.listing;
   307→      items.push({
   308→        id: it.id,
   309→        sortOrder: it.sortOrder,
   310→        type: "LISTING",
   311→        title: l.title,
   312→        slug: l.slug,
   313→        image: l.images[0]?.url ?? null,
   314→        categoryId: l.categoryId ?? null,
   315→        categoryName: l.category?.name ?? null,
   316→        brandId: l.brandId ?? null,
   317→        brandName: l.brand?.name ?? null,
   318→        modelId: l.modelId ?? null,
   319→        modelName: l.model?.name ?? null,
   320→        productId: l.productId ?? null,
   321→        year: l.year ?? null,
   322→        city: l.city ?? null,
   323→        province: l.province ?? null,
   324→        listingId: l.id,
   325→      });
   326→      listingIds.push(l.id);
   327→    } else if (it.brandId || it.modelId || it.productId) {
   328→      // Bare catalog item (no listing). Resolve the brand/model/product
   329→      // lazily so the comparison still has column metadata.
   330→      const [brand, model, product] = await Promise.all([
   331→        it.brandId
   332→          ? db.brand.findUnique({ where: { id: it.brandId }, select: { id: true, name: true, nameEn: true, country: true } })
   333→          : Promise.resolve(null),
   334→        it.modelId
   335→          ? db.productModel.findUnique({ where: { id: it.modelId }, select: { id: true, name: true, nameEn: true, categoryId: true, category: { select: { id: true, name: true } } } })
   336→          : Promise.resolve(null),
   337→        it.productId
   338→          ? db.product.findUnique({ where: { id: it.productId }, select: { id: true, canonicalName: true, slug: true, categoryId: true, category: { select: { id: true, name: true } } } })
   339→          : Promise.resolve(null),
   340→      ]);
   341→      items.push({
   342→        id: it.id,
   343→        sortOrder: it.sortOrder,
   344→        type: it.productId ? "PRODUCT" : it.modelId ? "MODEL" : "BRAND",
   345→        title: product?.canonicalName ?? model?.name ?? brand?.name ?? "مورد مقایسه",
   346→        slug: product?.slug ?? null,
   347→        image: null,
   348→        categoryId: product?.categoryId ?? model?.categoryId ?? null,
   349→        categoryName: product?.category?.name ?? model?.category?.name ?? null,
   350→        brandId: brand?.id ?? null,
   351→        brandName: brand?.name ?? null,
   352→        modelId: model?.id ?? null,
   353→        modelName: model?.name ?? null,
   354→        productId: product?.id ?? null,
   355→      });
   356→    }
   357→  }
   358→
   359→  if (items.length === 0) {
   360→    return {
   361→      items: [],
   362→      rows: [],
   363→      attributes: [],
   364→      differences: [],
   365→      crossCategoryWarning: false,
   366→      categories: [],
   367→    };
   368→  }
   369→
   370→  // ── Pull all ListingAttributeValue rows for the listings in scope. ──
   371→  // Group by listingId → Map<attributeId, value>
   372→  const attrValuesByListing = new Map<string, Map<string, any>>();
   373→  if (listingIds.length > 0) {
   374→    const rows = await db.listingAttributeValue.findMany({
   375→      where: { listingId: { in: listingIds } },
   376→      include: {
   377→        attribute: {
   378→          include: {
   379→            options: { orderBy: [{ sortOrder: "asc" }, { value: "asc" }] },
   380→          },
   381→        },
   382→      },
   383→    });
   384→    for (const v of rows) {
   385→      if (!attrValuesByListing.has(v.listingId)) {
   386→        attrValuesByListing.set(v.listingId, new Map());
   387→      }
   388→      attrValuesByListing.get(v.listingId)!.set(v.attributeId, v);
   389→    }
   390→  }
   391→
   392→  // Collect the union of attributes that appear on at least one listing
   393→  // (sorted by sortOrder for a stable column order).
   394→  const attributeMeta = new Map<string, any>();
   395→  for (const lm of attrValuesByListing.values()) {
   396→    for (const [attrId, v] of lm.entries()) {
   397→      if (!attributeMeta.has(attrId)) attributeMeta.set(attrId, v.attribute);
   398→    }
   399→  }
   400→
   401→  // ── COMPARE-ENGINE — admin can hide specific attributes via
   402→  // SiteSettings.compareVisibleAttributeIds (JSON array of attr IDs).
   403→  // When the field is null/empty, ALL attributes are shown (default).
   404→  // When set, ONLY those attribute IDs appear in the rows. ──
   405→  let visibleAttrIds: Set<string> | null = null;
   406→  try {
   407→    const settings = await db.siteSettings.findUnique({
   408→      where: { id: "main" },
   409→      select: { compareVisibleAttributeIds: true },
   410→    });
   411→    if (settings?.compareVisibleAttributeIds) {
   412→      const parsed = JSON.parse(settings.compareVisibleAttributeIds);
   413→      if (Array.isArray(parsed) && parsed.length > 0) {
   414→        visibleAttrIds = new Set(parsed.filter((x) => typeof x === "string"));
   415→      }
   416→    }
   417→  } catch {
   418→    /* ignore — fall back to showing all */
   419→  }
   420→
   421→  let sortedAttributes = Array.from(attributeMeta.values()).sort(
   422→    (a, b) => (a.sortOrder ?? 0) - (b.sortOrder ?? 0) || (a.name ?? "").localeCompare(b.name ?? ""),
   423→  );
   424→  if (visibleAttrIds && visibleAttrIds.size > 0) {
   425→    sortedAttributes = sortedAttributes.filter((a) => visibleAttrIds!.has(a.id));
   426→  }
   427→
   428→  // ── Fixed-spec rows (brand, model, year, hours, condition, price, city, etc.). ──
   429→  const rows: ComparisonRow[] = [];
   430→
   431→  const pushFixedRow = (
   432→    key: string,
   433→    label: string,
   434→    get: (it: ComparisonItemData) => { display: string; raw: string | null; normalized: string; unit?: string | null } | null,
   435→  ) => {
   436→    const cells: ComparisonCell[] = items.map((it) => {
   437→      const v = get(it);
   438→      if (!v || (v.display === "" && v.raw === null)) {
   439→        return { ...MISSING_CELL, itemId: it.id };
   440→      }
   441→      return {
   442→        itemId: it.id,
   443→        display: v.display || MISSING,
   444→        raw: v.raw,
   445→        normalized: v.normalized,
   446→        unit: v.unit ?? null,
   447→      };
   448→    });
   449→    const normalizeds = cells.map((c) => c.normalized);
   450→    const isDifferent =
   451→      normalizeds.filter((n) => n !== "").length >= 2 &&
   452→      new Set(normalizeds.filter((n) => n !== "")).size > 1;
   453→    rows.push({ key, label, category: "fixed", unit: null, cells, isDifferent });
   454→  };
   455→
   456→  pushFixedRow("brand", "برند", (it) =>
   457→    it.brandName ? { display: it.brandName, raw: it.brandName, normalized: it.brandName.toLowerCase() } : null,
   458→  );
   459→  pushFixedRow("model", "مدل", (it) =>
   460→    it.modelName ? { display: it.modelName, raw: it.modelName, normalized: it.modelName.toLowerCase() } : null,
   461→  );
   462→  pushFixedRow("category", "دسته‌بندی", (it) =>
   463→    it.categoryName ? { display: it.categoryName, raw: it.categoryName, normalized: it.categoryName.toLowerCase() } : null,
   464→  );
   465→  pushFixedRow("year", "سال ساخت", (it) =>
   466→    it.year ? { display: formatInt(it.year), raw: String(it.year), normalized: String(it.year) } : null,
   467→  );
   468→
   469→  // Hours — needs ListingAttributeValue lookup (since Listing.workingHours is also a column
   470→  // on Listing, prefer the listing's own column for V1).
   471→  pushFixedRow("hours", "ساعت کارکرد", (it) => {
   472→    if (!it.listingId) return null;
   473→    const listing = session.items.find((s) => s.id === it.id)?.listing;
   474→    const h = listing?.workingHours ?? null;
   475→    if (h == null) return null;
   476→    return { display: `${formatInt(h)} ساعت`, raw: String(h), normalized: String(h), unit: "ساعت" };
   477→  });
   478→  pushFixedRow("condition", "وضعیت", (it) => {
   479→    if (!it.listingId) return null;
   480→    const listing = session.items.find((s) => s.id === it.id)?.listing;
   481→    const c = listing?.condition ?? null;
   482→    if (!c) return null;
   483→    const label = CONDITION_LABELS[c] ?? c;
   484→    return { display: label, raw: c, normalized: c };
   485→  });
   486→  pushFixedRow("city", "شهر", (it) =>
   487→    it.city ? { display: it.city, raw: it.city, normalized: it.city.toLowerCase() } : null,
   488→  );
   489→  pushFixedRow("province", "استان", (it) =>
   490→    it.province ? { display: it.province, raw: it.province, normalized: it.province.toLowerCase() } : null,
   491→  );
   492→
   493→  // Price row — special: includes the estimated range from the price engine.
   494→  // We compute ranges in parallel below and merge them into the row.
   495→  const priceCells: ComparisonCell[] = items.map((it) => {
   496→    if (!it.listingId) {
   497→      // Bare catalog item (no listing) — no asking price.
   498→      return { ...MISSING_CELL, itemId: it.id };
   499→    }
   500→    const listing = session.items.find((s) => s.id === it.id)?.listing;
   501→    const price = listing?.price ?? null;
   502→    const priceType = listing?.priceType ?? "NEGOTIABLE";
   503→    if (price === null || price === undefined) {
   504→      const pt = PRICE_TYPE_LABELS[priceType] ?? MISSING;
   505→      return {
   506→        itemId: it.id,
   507→        display: pt,
   508→        raw: null,
   509→        normalized: pt,
   510→      };
   511→    }
   512→    const num = Number(price);
   513→    return {
   514→      itemId: it.id,
   515→      display: formatToman(num),
   516→      raw: String(num),
   517→      normalized: String(num),
   518→      unit: "تومان",
   519→    };
   520→  });
   521→
   522→  // Compute estimated ranges in parallel — only for items with a categoryId.
   523→  const rangeResults = await Promise.all(
   524→    items.map(async (it) => {
   525→      if (!it.categoryId) return null;
   526→      try {
   527→        const sug = await getPriceSuggestions({
   528→          categoryId: it.categoryId,
   529→          brandId: it.brandId ?? undefined,
   530→          year: it.year ?? undefined,
   531→        });
   532→        return sug;
   533→      } catch {
   534→        return null;
   535→      }
   536→    }),
   537→  );
   538→
   539→  const priceRanges: ComparisonPriceRange[] = items.map((it, idx) => {
   540→    const sug = rangeResults[idx];
   541→    if (!sug || sug.sampleSize === 0) {
   542→      return {
   543→        min: null,
   544→        max: null,
   545→        avg: null,
   546→        currency: "IRR",
   547→        confidence: "LOW" as const,
   548→        sampleSize: 0,
   549→      };
   550→    }
   551→    return {
   552→      min: sug.suggestedMin,
   553→      max: sug.suggestedMax,
   554→      avg: sug.suggestedAvg,
   555→      currency: "IRR",
   556→      confidence: sug.confidence,
   557→      sampleSize: sug.sampleSize,
   558→    };
   559→  });
   560→
   561→  // Build a richer display string for the price cell that includes
   562→  // the estimated range when available (so the diff detector still
   563→  // works on the raw asking price).
   564→  const priceCellsWithRange: ComparisonCell[] = priceCells.map((c, idx) => {
   565→    const r = priceRanges[idx];
   566→    if (!r || r.min === null || r.max === null) return c;
   567→    const rangeStr = ` (${formatToman(r.min)} – ${formatToman(r.max)})`;
   568→    return { ...c, display: c.display === MISSING ? MISSING : c.display + rangeStr };
   569→  });
   570→
   571→  const priceNorms = priceCells.map((c) => c.normalized);
   572→  const priceDiff =
   573→    priceNorms.filter((n) => n !== "" && n !== MISSING).length >= 2 &&
   574→    new Set(priceNorms.filter((n) => n !== "" && n !== MISSING)).size > 1;
   575→
   576→  rows.push({
   577→    key: "price",
   578→    label: "قیمت",
   579→    category: "fixed",
   580→    unit: "تومان",
   581→    cells: priceCellsWithRange,
   582→    isDifferent: priceDiff,
   583→    isPrice: true,
   584→    priceRanges,
   585→  });
   586→
   587→  // ── Dynamic attribute rows. ──
   588→  for (const attr of sortedAttributes) {
   589→    const key = `attr:${attr.id}`;
   590→    const label = attr.labelFa ?? attr.name ?? attr.nameEn ?? attr.key ?? "مشخصه";
   591→    const cells: ComparisonCell[] = items.map((it) => {
   592→      if (!it.listingId) return { ...MISSING_CELL, itemId: it.id };
   593→      const lm = attrValuesByListing.get(it.listingId);
   594→      const v = lm?.get(attr.id);
   595→      if (!v) return { ...MISSING_CELL, itemId: it.id };
   596→
   597→      // Resolve option label if it's a SELECT.
   598→      let display = "";
   599→      let raw: string | null = null;
   600→      let normalized = "";
   601→      const unit = v.unit ?? attr.unit ?? null;
   602→
   603→      if (v.optionId) {
   604→        const opt = attr.options.find((o: any) => o.id === v.optionId);
   605→        display = opt?.label ?? opt?.value ?? "";
   606→        raw = opt?.value ?? v.optionId ?? null;
   607→        normalized = String(opt?.value ?? v.optionId ?? "").toLowerCase();
   608→      } else if (v.numberValue !== null && v.numberValue !== undefined) {
   609→        const n = v.numberValue;
   610→        const num = formatToman(n);
   611→        // For non-currency numeric values, just Persian digits + unit.
   612→        if (attr.type === "CURRENCY" && n > 1000) {
   613→          display = num;
   614→        } else {
   615→          display = unit ? `${formatInt(n)} ${unit}` : formatInt(n);
   616→        }
   617→        raw = String(n);
   618→        normalized = String(n);
   619→      } else if (v.booleanValue !== null && v.booleanValue !== undefined) {
   620→        display = v.booleanValue ? "بله" : "خیر";
   621→        raw = String(v.booleanValue);
   622→        normalized = String(v.booleanValue);
   623→      } else if (v.dateValue) {
   624→        try {
   625→          const d = new Date(v.dateValue);
   626→          display = d.toLocaleDateString("fa-IR");
   627→          raw = d.toISOString();
   628→          normalized = d.toISOString().slice(0, 10);
   629→        } catch {
   630→          display = "";
   631→        }
   632→      } else if (v.textValue && v.textValue.trim() !== "") {
   633→        display = v.textValue;
   634→        raw = v.textValue;
   635→        normalized = v.textValue.trim().toLowerCase();
   636→      }
   637→
   638→      if (display === "" || display === null) {
   639→        return { ...MISSING_CELL, itemId: it.id };
   640→      }
   641→
   642→      return {
   643→        itemId: it.id,
   644→        display,
   645→        raw,
   646→        normalized,
   647→        unit: unit ?? null,
   648→        provenance: {
   649→          source: v.sourceType ?? null,
   650→          confidence: v.confidence ?? null,
   651→          verifiedAt: v.verifiedAt ? new Date(v.verifiedAt).toISOString() : null,
   652→          sourceReference: v.sourceReference ?? null,
   653→          verifiedBy: v.verifiedBy ?? null,
   654→        },
   655→      };
   656→    });
   657→
   658→    const normalizeds = cells.map((c) => c.normalized);
   659→    const isDifferent =
   660→      normalizeds.filter((n) => n !== "").length >= 2 &&
   661→      new Set(normalizeds.filter((n) => n !== "")).size > 1;
   662→
   663→    rows.push({
   664→      key,
   665→      label,
   666→      category: "attribute",
   667→      unit: attr.unit ?? null,
   668→      cells,
   669→      isDifferent,
   670→    });
   671→  }
   672→
   673→  // ── Cross-category warning. ──
   674→  const distinctCategories = new Map<string, string>();
   675→  for (const it of items) {
   676→    if (it.categoryId && it.categoryName && !distinctCategories.has(it.categoryId)) {
   677→      distinctCategories.set(it.categoryId, it.categoryName);
   678→    }
   679→  }
   680→  const crossCategoryWarning = distinctCategories.size > 1;
   681→
   682→  const attributes = rows.map((r) => r.key);
   683→  const differences = rows.filter((r) => r.isDifferent).map((r) => r.key);
   684→
   685→  return {
   686→    items,
   687→    rows,
   688→    attributes,
   689→    differences,
   690→    crossCategoryWarning,
   691→    categories: Array.from(distinctCategories.entries()).map(([id, name]) => ({ id, name })),
   692→  };
   693→}

   726→/**
   727→ * Generate a Persian AI summary of the comparison:
   728→ *   • key differences across items
   729→ *   • missing data
   730→ *   • price comparison
   731→ *
   732→ * No winner / no aggregate ranking is declared (spec §11 + §12).
   733→ *
   734→ * The summary is persisted on the session (aiSummary + aiSummaryAt)
   735→ * so subsequent reads of the same session don't re-bill the LLM.
   736→ */
   737→export async function generateAISummary(sessionId: string): Promise<string> {
   738→  const data = await getComparisonData(sessionId);
   739→  if (data.items.length < 2) {
   740→    return "حداقل دو مورد برای خلاصه‌سازی لازم است.";
   741→  }
   742→
   743→  // Build a compact JSON payload for the LLM. Cells are flattened to
   744→  // display strings to keep the prompt small.
   745→  const payload = {
   746→    items: data.items.map((it) => ({
   747→      title: it.title,
   748→      type: it.type,
   749→      brand: it.brandName ?? null,
   750→      model: it.modelName ?? null,
   751→      category: it.categoryName ?? null,
   752→      year: it.year ?? null,
   753→    })),
   754→    rows: data.rows.map((r) => ({
   755→      label: r.label,
   756→      isDifferent: r.isDifferent,
   757→      values: r.cells.map((c) => c.display),
   758→      priceRange: r.priceRanges
   759→        ? r.priceRanges.map((p) => ({
   760→            min: p.min,
   761→            max: p.max,
   762→            confidence: p.confidence,
   763→            sampleSize: p.sampleSize,
   764→          }))
   765→        : undefined,
   766→    })),
   767→    crossCategoryWarning: data.crossCategoryWarning,
   768→    differences: data.differences,
   769→  };
   770→
   771→  const prompt = `You are the HEAVIX Machine Comparison Assistant. Compare ${data.items.length} heavy-machinery items and write a CONCISE Persian (Farsi) summary. STRICT RULES:
   772→1. NEVER declare a winner, NEVER rank items, NEVER claim field-test results or performance guarantees.
   773→2. Summarize only DOCUMENTED differences from the provided data.
   774→3. Explicitly list MISSING data (fields shown as "—").
   775→4. Compare asking prices + estimated ranges when available — describe the price spread, NOT which is "better".
   776→5. If a cross-category warning is present, mention it.
   777→6. Output 3-5 short Persian bullet lines, each ≤ 30 words, prefixed with "• ".
   778→7. Do not include any English, headers, or markdown other than the bullet prefix.
   779→
   780→Data:
   781→${JSON.stringify(payload)}`;
   782→
   783→  let summary = "";
   784→  try {
   785→    const ZAIModule = await import("z-ai-web-dev-sdk");
   786→    const ZAI = (ZAIModule as any).default ?? ZAIModule;
   787→    const zai = await ZAI.create();
   788→    const completion = await zai.chat.completions.create({
   789→      messages: [
   790→        { role: "system", content: prompt },
   791→        { role: "user", content: "خلاصه را بنویس." },
   792→      ],
   793→      thinking: { type: "disabled" },
   794→    });
   795→    summary = (completion?.choices?.[0]?.message?.content ?? "").trim();
   796→  } catch (err) {
   797→    console.error("[compare-engine] AI summary failed:", err);
   798→    // Fall back to a deterministic local summary so the UI always has
   799→    // something useful to show.
   800→    summary = buildFallbackSummary(data);
   801→  }
   802→
   803→  if (!summary) {
   804→    summary = buildFallbackSummary(data);
   805→  }
   806→
   807→  // Persist the summary on the session for cached reads.
   808→  try {
   809→    await db.comparisonSession.update({
   810→      where: { id: sessionId },
   811→      data: { aiSummary: summary, aiSummaryAt: new Date() },
   812→    });
   813→  } catch (err) {
   814→    console.error("[compare-engine] persist AI summary failed:", err);
   815→  }
   816→
   817→  return summary;
   818→}

   820→/** Deterministic Persian summary used when the LLM is unavailable. */
   821→function buildFallbackSummary(data: ComparisonData): string {
   822→  const lines: string[] = [];
   823→  const diffRows = data.rows.filter((r) => r.isDifferent);
   824→  if (diffRows.length === 0) {
   825→    lines.push("• مشخصات موارد مقایسه یکسان است؛ تفاوت قابل توجهی مشاهده نشد.");
   826→  } else {
   827→    const top = diffRows.slice(0, 3);
   828→    for (const r of top) {
   829→      const values = r.cells.map((c) => c.display).filter((d) => d !== MISSING);
   830→      if (values.length >= 2) {
   831→        lines.push(`• ${r.label}: ${values.slice(0, 3).join("، ")}.`);
   832→      }
   833→    }
   834→  }
   835→  // Missing data
   836→  const missingRows = data.rows.filter((r) => r.cells.some((c) => c.display === MISSING));
   837→  if (missingRows.length > 0) {
   838→    lines.push(`• داده‌های ناقص: ${missingRows.slice(0, 4).map((r) => r.label).join("، ")}.`);
   839→  }
   840→  // Price
   841→  const priceRow = data.rows.find((r) => r.isPrice);
   842→  if (priceRow && priceRow.priceRanges) {
   843→    const withRange = priceRow.priceRanges.filter((p) => p.min !== null && p.max !== null);
   844→    if (withRange.length > 0) {
   845→      lines.push(`• بازه تخمینی قیمت: ${withRange.length} مورد بر اساس داده بازار.`);
   846→    }
   847→  }
   848→  if (data.crossCategoryWarning) {
   849→    lines.push(`• هشدار: موارد از دسته‌های متفاوت هستند (${data.categories.map((c) => c.name).join("، ")}).`);
   850→  }
   851→  lines.push("• هویکس برنده اعلام نمی‌کند؛ تصمیم‌گیری نهایی با کاربر است.");
   852→  return lines.join("\n");
   853→}
