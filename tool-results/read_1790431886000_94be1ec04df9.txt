    63→function listRouteFiles(): string[] {
    64→  const results: string[] = [];
    65→  function walk(dir: string) {
    66→    const entries = fs.readdirSync(dir, { withFileTypes: true });
    67→    for (const entry of entries) {
    68→      const full = path.join(dir, entry.name);
    69→      if (entry.isDirectory()) walk(full);
    70→      else if (entry.name === 'route.ts') results.push(full);
    71→    }
    72→  }
    73→  walk(STORE_ROUTES_DIR);
    74→  // Exclude health endpoints (monitoring stubs — no auth required)
    75→  return results.filter(f => !f.includes('/health/')).sort();
    76→}
    77→
    78→function readRoute(rel: string): string {
    79→  return fs.readFileSync(rel, 'utf8');
    80→}
    81→
    82→function listServiceFiles(): string[] {
    83→  return STORE_SERVICE_FILES.filter(f => fs.existsSync(f));
    84→}
    85→
    86→function readService(rel: string): string {
    87→  return fs.readFileSync(rel, 'utf8');
    88→}
    89→
    90→// Combined scan: returns content of every route file + every service file.
    91→// Used for audit/mutation/action-key contracts (which now live in both).
    92→function readAllStoreFiles(): { path: string; content: string }[] {
    93→  const routes = listRouteFiles().map(p => ({ path: p, content: readRoute(p) }));
    94→  const services = listServiceFiles().map(p => ({ path: p, content: readService(p) }));
    95→  return [...routes, ...services];
    96→}
