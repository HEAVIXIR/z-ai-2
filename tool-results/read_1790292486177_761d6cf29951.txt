     4→/* ============================================================
     5→   HEAVIX — Price Estimation Engine (PRICE-ENGINE)
     6→   ------------------------------------------------------------
     7→   Implements the data-driven price estimation pipeline described
     8→   in `docs/HEAVIX-PRICE-ESTIMATION-SPEC-V1.0.md`:
     9→
    10→     Raw Market Data
    11→     → Comparable Selection (same brand+category, year ±3,
    12→       working-hours ±30%)
    13→     → Feature Normalization (Toman/IRR; no fabricated values)
    14→     → Price Estimation (median, p25, p75)
    15→     → Confidence / Data Quality (count-based + variance)
    16→     → Explanation (main drivers + warnings)
    17→
    18→   This module is INTENTIONALLY SERVER-ONLY — it imports the Prisma
    19→   client directly. NEVER import it from a Client Component.
    20→
    21→   Spec rules enforced here:
    22→     • Asking vs Estimated are NEVER conflated — `askingPrice`
    23→       is read from `Listing.price`; `estimatedPrice` is computed.
    24→     • Confidence is NEVER presented as a guarantee.
    25→     • No price is invented without source data — when there are
    26→       fewer than 2 comparables the engine returns
    27→       `confidence = "INSUFFICIENT"` and no estimatedPrice.
    28→     • Overrides are audited via the shared `logAudit` helper.
    29→   ============================================================ */

   125→/* ─────────── Private statistical helpers ─────────── */
   126→
   127→function median(sorted: number[]): number | null {
   128→  if (sorted.length === 0) return null;
   129→  const mid = Math.floor(sorted.length / 2);
   130→  return sorted.length % 2 === 0
   131→    ? (sorted[mid - 1]! + sorted[mid]!) / 2
   132→    : sorted[mid]!;
   133→}
   134→
   135→function percentile(sorted: number[], p: number): number | null {
   136→  if (sorted.length === 0) return null;
   137→  const idx = (sorted.length - 1) * p;
   138→  const lo = Math.floor(idx);
   139→  const hi = Math.ceil(idx);
   140→  if (lo === hi) return sorted[lo] ?? null;
   141→  const frac = idx - lo;
   142→  const a = sorted[lo]!;
   143→  const b = sorted[hi]!;
   144→  return a + (b - a) * frac;
   145→}
   146→
   147→/** Coefficient of variation (σ / μ). Robust spread indicator. */
   148→function coefficientOfVariation(values: number[]): number | null {
   149→  if (values.length < 2) return null;
   150→  const mean = values.reduce((s, v) => s + v, 0) / values.length;
   151→  if (mean === 0) return null;
   152→  const variance =
   153→    values.reduce((s, v) => s + (v - mean) ** 2, 0) / values.length;
   154→  return Math.sqrt(variance) / mean;
   155→}
   156→
   157→function confidenceForCount(count: number): ConfidenceLevel {
   158→  if (count >= 8) return "HIGH";
   159→  if (count >= 4) return "MEDIUM";
   160→  if (count >= 2) return "LOW";
   161→  return "INSUFFICIENT";
   162→}
   163→
   164→function freshnessForLatestObserved(latest: Date | null): DataFreshness | null {
   165→  if (!latest) return null;
   166→  const days = (Date.now() - latest.getTime()) / (1000 * 60 * 60 * 24);
   167→  if (days <= FRESH_DAYS) return "FRESH";
   168→  if (days <= RECENT_DAYS) return "RECENT";
   169→  return "STALE";
   170→}

   172→/* ─────────── Core: estimatePrice ─────────── */
   173→
   174→export async function estimatePrice(
   175→  params: EstimateInput,
   176→): Promise<EstimateResult> {
   177→  const drivers: string[] = [];
   178→  const warnings: string[] = [];
   179→
   180→  /* 1. Resolve target attributes — either from a listingId or
   181→        raw params. NEVER invent missing fields. */
   182→  let target: {
   183→    brandId: string | null;
   184→    categoryId: string | null;
   185→    modelId: string | null;
   186→    year: number | null;
   187→    hours: number | null;
   188→    condition: string | null;
   189→    city: string | null;
   190→  };
   191→
   192→  if (params.listingId) {
   193→    const listing = await db.listing.findUnique({
   194→      where: { id: params.listingId },
   195→      select: {
   196→        id: true ,
   197→        brandId: true,
   198→        categoryId: true,
   199→        modelId: true,
   200→        year: true,
   201→        workingHours: true,
   202→        condition: true,
   203→        city: true,
   204→      },
   205→    });
   206→    if (!listing) {
   207→      return emptyEstimate(["آگهی یافت نشد"]);
   208→    }
   209→    target = {
   210→      brandId: params.brandId ?? listing.brandId,
   211→      categoryId: params.categoryId ?? listing.categoryId,
   212→      modelId: params.modelId ?? listing.modelId,
   213→      year: params.year ?? listing.year,
   214→      hours: params.hours ?? listing.workingHours,
   215→      condition: params.condition ?? listing.condition,
   216→      city: params.city ?? listing.city,
   217→    };
   218→  } else {
   219→    target = {
   220→      brandId: params.brandId ?? null,
   221→      categoryId: params.categoryId ?? null,
   222→      modelId: params.modelId ?? null,
   223→      year: params.year ?? null,
   224→      hours: params.hours ?? null,
   225→      condition: params.condition ?? null,
   226→      city: params.city ?? null,
   227→    };
   228→  }
   229→
   230→  /* 2. Comparable selection — require brand+category at minimum. */
   231→  if (!target.brandId || !target.categoryId) {
   232→    return emptyEstimate([
   233→      "برای تخمین قیمت، برند و دسته‌بندی لازم است.",
   234→    ]);
   235→  }
   236→
   237→  const where: {
   238→    status: string;
   239→    price: { not: null };
   240→    brandId: string;
   241→    categoryId: string;
   242→    modelId?: string;
   243→    year?: { gte: number; lte: number };
   244→    workingHours?: { gte: number; lte: number };
   245→    NOT?: { id: string }[];
   246→  } = {
   247→    status: "PUBLISHED",
   248→    price: { not: null },
   249→    brandId: target.brandId,
   250→    categoryId: target.categoryId,
   251→  };
   252→  if (target.modelId) where.modelId = target.modelId;
   253→  if (params.listingId) (where as any).NOT = { id: params.listingId };
   254→
   255→  // Year window (±3). Apply only if target year is known.
   256→  if (target.year && Number.isFinite(target.year)) {
   257→    where.year = {
   258→      gte: target.year - YEAR_TOLERANCE,
   259→      lte: target.year + YEAR_TOLERANCE,
   260→    };
   261→  }
   262→
   263→  // Working-hours window (±30%). Apply only if target hours is known.
   264→  if (target.hours != null && Number.isFinite(target.hours) && target.hours > 0) {
   265→    const delta = Math.max(1, target.hours * HOURS_TOLERANCE_RATIO);
   266→    where.workingHours = {
   267→      gte: Math.floor(target.hours - delta),
   268→      lte: Math.ceil(target.hours + delta),
   269→    };
   270→  }
   271→
   272→  // Always also consider PriceObservations of the same brand+category
   273→  // (source = LISTING) so the engine benefits from historical data even
   274→  // when active comparable listings are scarce. NEVER fabricate prices.
   275→  const obsRows = await db.priceObservation.findMany({
   276→    where: {
   277→      brandId: target.brandId,
   278→      categoryId: target.categoryId,
   279→      status: "ACTIVE",
   280→      askingPrice: { not: null },
   281→    },
   282→    select: {
   283→      id: true ,
   284→      askingPrice: true,
   285→      observedAt: true,
   286→      listingId: true,
   287→    },
   288→    take: MAX_COMPARABLES,
   289→    orderBy: { observedAt: "desc" },
   290→  });
   291→
   292→  // ── Fallback: if the strict query (year±3, hours±30%) yields
   293→  //    fewer than 2 active comparables, retry with a relaxed query
   294→  //    (drop the hours window and widen year to ±5) so the engine
   295→  //    still produces an estimate on smaller datasets. The relaxed
   296→  //    pass is flagged via a warning so consumers know similarity
   297→  //    is looser. NEVER fabricates prices — only uses real listings.
   298→  let rows = await db.listing.findMany({
   299→    where,
   300→    select: {
   301→      id: true ,
   302→      title: true,
   303→      slug: true,
   304→      price: true,
   305→      year: true,
   306→      workingHours: true,
   307→      city: true,
   308→      createdAt: true,
   309→    },
   310→    take: MAX_COMPARABLES,
   311→    orderBy: { createdAt: "desc" },
   312→  });
   313→
   314→  let relaxed = false;
   315→  if (rows.length < 2) {
   316→    const relaxedWhere: typeof where = { ...where };
   317→    delete (relaxedWhere ).workingHours;
   318→    if (target.year && Number.isFinite(target.year)) {
   319→      relaxedWhere.year = {
   320→        gte: target.year - (YEAR_TOLERANCE + 2),
   321→        lte: target.year + (YEAR_TOLERANCE + 2),
   322→      };
   323→    } else {
   324→      delete (relaxedWhere ).year;
   325→    }
   326→    const relaxedRows = await db.listing.findMany({
   327→      where: relaxedWhere,
   328→      select: {
   329→        id: true ,
   330→        title: true,
   331→        slug: true,
   332→        price: true,
   333→        year: true,
   334→        workingHours: true,
   335→        city: true,
   336→        createdAt: true,
   337→      },
   338→      take: MAX_COMPARABLES,
   339→      orderBy: { createdAt: "desc" },
   340→    });
   341→    if (relaxedRows.length > rows.length) {
   342→      rows = relaxedRows;
   343→      relaxed = true;
   344→    }
   345→  }
   346→
   347→  // ── Final fallback: brand+category only (no year/hours window).
   348→  //    Keeps the engine usable on small catalogs — every comparable
   349→  //    is a real listing, we just broaden similarity to the maximum.
   350→  let brandCategoryOnly = false;
   351→  if (rows.length < 2) {
   352→    const broadRows = await db.listing.findMany({
   353→      where: {
   354→        status: "PUBLISHED",
   355→        price: { not: null },
   356→        brandId: target.brandId,
   357→        categoryId: target.categoryId,
   358→        ...(params.listingId ? { NOT: { id: params.listingId } } : {}),
   359→      },
   360→      select: {
   361→        id: true ,
   362→        title: true,
   363→        slug: true,
   364→        price: true,
   365→        year: true,
   366→        workingHours: true,
   367→        city: true,
   368→        createdAt: true,
   369→      },
   370→      take: MAX_COMPARABLES,
   371→      orderBy: { createdAt: "desc" },
   372→    });
   373→    if (broadRows.length > rows.length) {
   374→      rows = broadRows;
   375→      brandCategoryOnly = true;
   376→    }
   377→  }
   378→
   379→
   380→  // Merge: prices from active listings + prices from observations.
   381→  // Each row in each table is treated as a DISTINCT market observation
   382→  // (spec §14 — every observation has its own provenance). Multiple
   383→  // observations of the same listing over time legitimately count as
   384→  // separate comparables — this is how the engine benefits from
   385→  // historical price drift on the same machine.
   386→  const seenIds = new Set<string>();
   387→  type Comp = {
   388→    id: string;
   389→    title: string;
   390→    slug: string;
   391→    price: number;
   392→    year: number | null;
   393→    workingHours: number | null;
   394→    city: string | null;
   395→    observedAt: Date;
   396→    similarity: number;
   397→  };
   398→  const comps: Comp[] = [];
   399→
   400→  for (const r of rows) {
   401→    if (!r.price) continue;
   402→    if (seenIds.has(r.id)) continue;
   403→    seenIds.add(r.id);
   404→    comps.push({
   405→      id: r.id,
   406→      title: r.title,
   407→      slug: r.slug,
   408→      price: Number(r.price),
   409→      year: r.year,
   410→      workingHours: r.workingHours,
   411→      city: r.city,
   412→      observedAt: r.createdAt,
   413→      similarity: computeSimilarity(target, {
   414→        year: r.year,
   415→        hours: r.workingHours,
   416→        city: r.city,
   417→      }),
   418→    });
   419→  }
   420→  for (const o of obsRows) {
   421→    if (!o.askingPrice) continue;
   422→    if (seenIds.has(o.id)) continue;
   423→    seenIds.add(o.id);
   424→    comps.push({
   425→      id: o.id,
   426→      title: "(مشاهده بازار)",
   427→      slug: "",
   428→      price: Number(o.askingPrice),
   429→      year: null,
   430→      workingHours: null,
   431→      city: null,
   432→      observedAt: o.observedAt,
   433→      similarity: 0.5,
   434→    });
   435→  }
   436→
   437→  // ── STEP 6B.1: PriceRecord fold-in REMOVED.
   438→  // The legacy PriceRecord table is deprecated (canonical model is
   439→  // PriceObservation per PHASE-6A-RECONCILE). The PriceRecord writer
   440→  // (recordPriceFromListing) is dead code with 0 callers, so the table
   441→  // is empty — folding in its rows would yield zero comparables anyway.
   442→  // All comparable selection now uses PriceObservation exclusively.
   443→
   444→  if (comps.length < 2) {
   445→    return emptyEstimate([
   446→      "داده مشابه کافی برای تخمین قیمت موجود نیست.",
   447→      "این تخمین جایگزین کارشناسی حضوری نیست.",
   448→    ]);
   449→  }
   450→
   451→  // Sort by similarity (desc) for the comparables preview.
   452→  const bySimilarity = [...comps].sort((a, b) => b.similarity - a.similarity);
   453→
   454→  // Price statistics computed across ALL comparables.
   455→  const prices = comps.map((c) => c.price).sort((a, b) => a - b);
   456→  const med = median(prices);
   457→  const p25 = percentile(prices, 0.25);
   458→  const p75 = percentile(prices, 0.75);
   459→  const count = comps.length;
   460→  const confidence = confidenceForCount(count);
   461→
   462→  // Latest observation date → freshness.
   463→  const latest = comps.reduce<Date | null>((acc, c) => {
   464→    if (!acc || c.observedAt > acc) return c.observedAt;
   465→    return acc;
   466→  }, null);
   467→  const dataFreshness = freshnessForLatestObserved(latest);
   468→
   469→  // Drivers (explain what influenced the estimate).
   470→  if (target.brandId) drivers.push("برند");
   471→  if (target.categoryId) drivers.push("دسته‌بندی");
   472→  if (target.modelId) drivers.push("مدل");
   473→  if (target.year) drivers.push("سال ساخت");
   474→  if (target.hours != null) drivers.push("ساعت کارکرد");
   475→  if (target.condition) drivers.push("وضعیت دستگاه");
   476→  if (target.city) drivers.push("موقعیت جغرافیایی");
   477→  drivers.push(`تعداد موارد مشابه: ${count}`);
   478→
   479→  // Warnings.
   480→  const cv = coefficientOfVariation(prices);
   481→  if (cv != null && cv > 0.5) {
   482→    warnings.push("پراکندگی بالای قیمت در موارد مشابه — تخمین با احتیاط ببینید.");
   483→  }
   484→  if (dataFreshness === "STALE") {
   485→    warnings.push("آخرین داده‌های مشاهده‌شده قدیمی هستند (بیش از ۹۰ روز).");
   486→  }
   487→  if (confidence === "LOW") {
   488→    warnings.push("تعداد موارد مشابه کم است؛ تخمین با قطعیت بالا تلقی نشود.");
   489→  }
   490→  if (confidence === "INSUFFICIENT") {
   491→    warnings.push("داده ناکافی — تخمین معتبر نیست.");
   492→  }
   493→  if (relaxed) {
   494→    warnings.push(
   495→      "بازه جستجوی موارد مشابه به دلیل کمبود داده گسترش یافت (تخمین با دقت کمتر).",
   496→    );
   497→  }
   498→  if (brandCategoryOnly) {
   499→    warnings.push(
   500→      "به دلیل کمبود داده، تخمین بر اساس تمام آگهی‌های همان برند و دسته‌بندی محاسبه شده است (بدون فیلتر سال/کارکرد).",
   501→    );
   502→  }
   503→
   504→  // Range: use p25/p75 (when available), widened for low confidence.
   505→  let lower = p25 ?? null;
   506→  let upper = p75 ?? null;
   507→  if (med != null) {
   508→    if (lower == null) lower = med * 0.85;
   509→    if (upper == null) upper = med * 1.15;
   510→    if (confidence === "LOW") {
   511→      lower = med * 0.8;
   512→      upper = med * 1.2;
   513→    } else if (confidence === "MEDIUM") {
   514→      lower = med * 0.88;
   515→      upper = med * 1.12;
   516→    } else if (confidence === "HIGH") {
   517→      lower = med * 0.92;
   518→      upper = med * 1.08;
   519→    }
   520→  }
   521→
   522→  return {
   523→    estimatedPrice: med,
   524→    priceLower: lower,
   525→    priceUpper: upper,
   526→    medianPrice: med,
   527→    confidence,
   528→    comparableCount: count,
   529→    dataFreshness,
   530→    mainDrivers: drivers,
   531→    warnings,
   532→    modelVersion: PRICE_MODEL_VERSION,
   533→    comparables: bySimilarity.slice(0, 12).map((c) => ({
   534→      id: c.id,
   535→      title: c.title,
   536→      slug: c.slug,
   537→      price: c.price,
   538→      year: c.year,
   539→      workingHours: c.workingHours,
   540→      city: c.city,
   541→      similarity: c.similarity,
   542→    })),
   543→  };
   544→}

   552→    confidence: "INSUFFICIENT",

   554→    dataFreshness: null,
   555→    mainDrivers: [],
   556→    warnings,
   557→    modelVersion: PRICE_MODEL_VERSION,
   558→    comparables: [],
   559→  };

   562→/** Heuristic similarity score (0..1) — higher is closer. */
   563→function computeSimilarity(
   564→  target: { year: number | null; hours: number | null; city: string | null },
   565→  cand: { year: number | null; hours: number | null; city: string | null },
   566→): number {
   567→  let score = 0.5;
   568→  if (target.year != null && cand.year != null) {
   569→    const d = Math.abs(target.year - cand.year);
   570→    score += Math.max(0, 0.25 - d * 0.05);
   571→  }
   572→  if (target.hours != null && cand.hours != null && target.hours > 0) {
   573→    const ratio = Math.abs(target.hours - cand.hours) / target.hours;
   574→    score += Math.max(0, 0.15 - ratio * 0.2);
   575→  }
   576→  if (target.city && cand.city && target.city === cand.city) {
   577→    score += 0.1;
   578→  }
   579→  return Math.min(1, score);
   580→}

   582→/* ─────────── getPriceHealth ─────────── */
   583→
   584→export async function getPriceHealth(
   585→  listingId: string,
   586→): Promise<PriceHealthResult> {
   587→  const listing = await db.listing.findUnique({
   588→    where: { id: listingId },
   589→    select: {
   590→      id: true ,
   591→      price: true,
   592→      brandId: true,
   593→      categoryId: true,
   594→      modelId: true,
   595→      year: true,
   596→      workingHours: true,
   597→      condition: true,
   598→      city: true,
   599→    },
   600→  });
   601→  if (!listing) {
   602→    return {
   603→      status: "INSUFFICIENT",
   604→      askingPrice: null,
   605→      estimatedLower: null,
   606→      estimatedUpper: null,
   607→      estimatedPrice: null,
   608→      confidence: null,
   609→      deviationPct: null,
   610→    };
   611→  }
   612→
   613→  const estimate = await estimatePrice({
   614→    listingId: listing.id,
   615→    brandId: listing.brandId,
   616→    categoryId: listing.categoryId,
   617→    modelId: listing.modelId,
   618→    year: listing.year,
   619→    hours: listing.workingHours,
   620→    condition: listing.condition,
   621→    city: listing.city,
   622→  });
   623→
   624→  const asking =
   625→    listing.price != null ? Number(listing.price) : null;
   626→
   627→  if (
   628→    asking == null ||
   629→    estimate.estimatedPrice == null ||
   630→    estimate.priceLower == null ||
   631→    estimate.priceUpper == null ||
   632→    estimate.confidence === "INSUFFICIENT"
   633→  ) {
   634→    return {
   635→      status: "INSUFFICIENT",
   636→      askingPrice: asking,
   637→      estimatedLower: estimate.priceLower,
   638→      estimatedUpper: estimate.priceUpper,
   639→      estimatedPrice: estimate.estimatedPrice,
   640→      confidence: estimate.confidence,
   641→      deviationPct: null,
   642→    };
   643→  }
   644→
   645→  let status: PriceHealthStatus;
   646→  if (asking < estimate.priceLower) status = "BELOW_RANGE";
   647→  else if (asking > estimate.priceUpper) status = "ABOVE_RANGE";
   648→  else status = "IN_RANGE";
   649→
   650→  const deviationPct =
   651→    estimate.estimatedPrice > 0
   652→      ? (asking - estimate.estimatedPrice) / estimate.estimatedPrice
   653→      : 0;
   654→
   655→  return {
   656→    status,
   657→    askingPrice: asking,
   658→    estimatedLower: estimate.priceLower,
   659→    estimatedUpper: estimate.priceUpper,
   660→    estimatedPrice: estimate.estimatedPrice,
   661→    confidence: estimate.confidence,
   662→    deviationPct,
   663→  };
   664→}

   718→/* ─────────── createOverride (admin, audited) ─────────── */
   719→
   720→export async function createOverride(args: {
   721→  listingId: string;
   722→  overridePrice: number;
   723→  reason: string;
   724→  adminId?: string | null;
   725→  ip?: string | null;
   726→  userAgent?: string | null;
   727→}): Promise<{ id: string }> {
   728→  if (!args.listingId) throw new Error("listingId is required");
   729→  if (!Number.isFinite(args.overridePrice) || args.overridePrice <= 0) {
   730→    throw new Error("overridePrice must be a positive finite number");
   731→  }
   732→  if (!args.reason || args.reason.trim().length < 3) {
   733→    throw new Error("reason is required (min 3 chars)");
   734→  }
   735→
   736→  // Compute the current estimate so we can record the "before" value.
   737→  const estimate = await estimatePrice({ listingId: args.listingId });
   738→  const originalEstimate = estimate.estimatedPrice ?? 0;
   739→
   740→  const listing = await db.listing.findUnique({
   741→    where: { id: args.listingId },
   742→    select: { id: true , title: true, slug: true, price: true },
   743→  });
   744→
   745→  const created = await db.priceOverride.create({
   746→    data: {
   747→      listingId: args.listingId,
   748→      originalEstimate,
   749→      overridePrice: args.overridePrice,
   750→      reason: args.reason.trim(),
   751→      overriddenBy: args.adminId ?? "ADMIN",
   752→    },
   753→  });
   754→
   755→  await logAudit({
   756→    actorId: args.adminId ?? null,
   757→    actorType: "ADMIN",
   758→    action: "pricing.override",
   759→    entityType: "PriceOverride",
   760→    entityId: created.id,
   761→    before: {
   762→      listingId: args.listingId,
   763→      listingTitle: listing?.title ?? null,
   764→      originalEstimate,
   765→      listingAskingPrice: listing?.price != null ? Number(listing.price) : null,
   766→    },
   767→    after: {
   768→      listingId: args.listingId,
   769→      overridePrice: args.overridePrice,
   770→      reason: args.reason.trim(),
   771→    },
   772→    ip: args.ip ?? null,
   773→    userAgent: args.userAgent ?? null,
   774→    reason: args.reason.trim(),
   775→  });
   776→
   777→  return { id: created.id };
   778→}
